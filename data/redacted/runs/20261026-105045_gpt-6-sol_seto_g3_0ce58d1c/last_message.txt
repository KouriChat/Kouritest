def point_in_polygon(lon, lat, polygon):
    """Ray-casting point test; boundary points are treated as land."""
    inside = False
    count = len(polygon)
    for i in range(count):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % count]

        # Count points on an edge as inside.
        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-12:
            if (min(x1, x2) - 1e-12 <= lon <= max(x1, x2) + 1e-12
                    and min(y1, y2) - 1e-12 <= lat <= max(y1, y2) + 1e-12):
                return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < crossing_lon:
                inside = not inside
    return inside


def is_land(lon, lat):
    land = any(point_in_polygon(lon, lat, poly) for poly in LAND_POLYGONS)
    if land and any(point_in_polygon(lon, lat, poly) for poly in WATER_POLYGONS):
        return 0
    return int(land)


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    # 64 rows from 35N to 33N; 253 columns from 129E to 137E.
    step = 2.0 / 63.0
    matrix = np.zeros((64, 253), dtype=np.uint8)
    for row in range(64):
        lat = 35.0 - row * step
        for col in range(253):
            lon = 129.0 + col * step
            matrix[row, col] = is_land(lon, lat)

    with (output_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # White is land; black is water. No filtering or shoreline post-processing.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((2024, 512), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    assert matrix.shape == (64, 253)
    assert matrix.size == 16192
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (253, 64)
    assert enlarged.size == (2024, 512)


if __name__ == "__main__":
    main()
