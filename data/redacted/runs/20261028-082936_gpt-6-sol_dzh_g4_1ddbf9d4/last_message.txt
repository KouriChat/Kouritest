def point_in_polygon(lon, lat, polygon):
    """Ray-casting point test; boundary points are treated as land."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        cross = (lon - xi) * (yj - yi) - (lat - yi) * (xj - xi)
        if abs(cross) < 1e-10 and min(xi, xj) <= lon <= max(xi, xj) \
                and min(yi, yj) <= lat <= max(yi, yj):
            return True
        if (yi > lat) != (yj > lat):
            x_cross = xi + (lat - yi) * (xj - xi) / (yj - yi)
            if lon < x_cross:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    return any(
        point_in_polygon(lon, lat, outline)
        for outline in (IBERIA, EUROPE, ANATOLIA, AFRICA, *ISLANDS)
    )


def main():
    latitudes = 45.0 - 0.2 * np.arange(71)
    longitudes = -9.0 + 0.2 * np.arange(231)
    matrix = np.zeros((71, 231), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(is_land(float(lon), float(lat)))

    assert matrix.shape == (71, 231)
    assert matrix.size == 16401
    assert np.isin(matrix, (0, 1)).all()

    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    with (output_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Matrix values are 1=land (white) and 0=water (black).
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((231 * 8, 71 * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    assert image.size == (231, 71)
    assert enlarged.size == (1848, 568)
    assert set(image.getdata()) <= {0, 255}
