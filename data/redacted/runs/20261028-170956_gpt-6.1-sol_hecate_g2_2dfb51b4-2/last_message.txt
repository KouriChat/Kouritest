本次无法实际运行脚本或展示生成后的图片；以下是可独立运行的生成器。它用手写的近似经纬度多边形表示海岸线与较大的岛屿，再按每个采样点是否落入陆地多边形生成矩阵；这不是逐点独立推理，细小岛礁也未必能完整还原。依赖：`numpy`、`Pillow`。

def point_on_segment(x, y, ax, ay, bx, by, eps=1e-10):
    cross = (x - ax) * (by - ay) - (y - ay) * (bx - ax)
    if abs(cross) > eps:
        return False
    return (min(ax, bx) - eps <= x <= max(ax, bx) + eps
            and min(ay, by) - eps <= y <= max(ay, by) + eps)


def inside_polygon(lon, lat, polygon):
    """Ray casting; boundary points are classified as land."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if point_on_segment(lon, lat, xi, yi, xj, yj):
            return True
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def main():

    output_dir.mkdir(parents=True, exist_ok=True)

    # The requested interval is 2/126 degrees, yielding 127 samples per axis.
    latitudes = np.linspace(55.0, 53.0, 127)
    longitudes = np.linspace(-131.0, -129.0, 127)
    land_polygons = [MAINLAND, GRAHAM, MORESBY, *ISLANDS]

    matrix = np.zeros((127, 127), dtype=np.uint8)
    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(
                any(inside_polygon(float(lon), float(lat), polygon)
                    for polygon in land_polygons)
            )

    with (output_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Land is white; water is black.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((1016, 1016), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    assert matrix.shape == (127, 127)
    assert matrix.size == 16129
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (127, 127)
    assert enlarged.size == (1016, 1016)
    assert set(image.getdata()).issubset({(0, 0, 0), (255, 255, 255)})


if __name__ == "__main__":
    main()
