def contains(polygon, x, y):
    """Ray casting; x is longitude and y is latitude."""
    inside = False
    x1, y1 = polygon[-1]
    for x2, y2 in polygon:
        if (y1 > y) != (y2 > y):
            crossing = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing:
                inside = not inside
        x1, y1 = x2, y2
    return inside

    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)

    latitudes = list(range(89, -90, -2))
    longitudes = list(range(-179, 180, 2))
    matrix = [
        [int(any(contains(poly, lon, lat) for poly in POLYGONS))
         for lon in longitudes]
        for lat in latitudes
    ]

    assert len(matrix) == 90
    assert all(len(row) == 180 for row in matrix)
    assert all(value in (0, 1) for row in matrix for value in row)
    assert sum(len(row) for row in matrix) == 16200

    csv_path = output / "candidate.csv"
    with csv_path.open("w", encoding="ascii", newline="") as handle:
        csv.writer(handle, lineterminator="\n").writerows(matrix)

    image = Image.new("L", (180, 90))
    image.putdata([255 * value for row in matrix for value in row])
    image.save(output / "candidate.png")

    try:
        nearest = Image.Resampling.NEAREST
    except AttributeError:
        nearest = Image.NEAREST
    enlarged = image.resize((1440, 720), resample=nearest)
    enlarged.save(output / "candidate_8x.png")

    # Check the written files, including every pixel in each 8x8 block.
    with csv_path.open("r", encoding="ascii", newline="") as handle:
        rows = list(csv.reader(handle))
    assert len(rows) == 90
    assert all(len(row) == 180 for row in rows)
    assert all(value in ("0", "1") for row in rows for value in row)

    with Image.open(output / "candidate.png") as original, \
         Image.open(output / "candidate_8x.png") as large:
        assert original.size == (180, 90)
        assert large.size == (1440, 720)
        assert set(original.getdata()) <= {0, 255}
        assert set(large.getdata()) <= {0, 255}
        small_pixels = original.load()
        large_pixels = large.load()
        for y in range(720):
            for x in range(1440):
                assert large_pixels[x, y] == small_pixels[x // 8, y // 8]
        for y, row in enumerate(rows):
            for x, value in enumerate(row):
                assert small_pixels[x, y] == 255 * int(value)

    print("Validated: 90 x 180 = 16,200 binary values")
    print("Validated: 180 x 90 and 1440 x 720 black/white PNGs")
    print("Validated: exact 8x nearest-neighbor enlargement")
    print("Files written to:", output.resolve())


if __name__ == "__main__":
    main()
