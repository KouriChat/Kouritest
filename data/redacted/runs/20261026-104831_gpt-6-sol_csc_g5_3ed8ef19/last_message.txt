def on_segment(x, y, a, b, eps=1e-10):
    cross = (x - a[0]) * (b[1] - a[1]) - (y - a[1]) * (b[0] - a[0])
    if abs(cross) > eps:
        return False
    return (min(a[0], b[0]) - eps <= x <= max(a[0], b[0]) + eps
            and min(a[1], b[1]) - eps <= y <= max(a[1], b[1]) + eps)


def inside_polygon(x, y, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if on_segment(x, y, (xi, yi), (xj, yj)):
            return True
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    land = any(inside_polygon(lon, lat, p) for p in LAND_POLYGONS)
    if land and any(inside_polygon(lon, lat, p) for p in WATER_POLYGONS):
        land = False
    return land


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    latitudes = [23.0 - 0.2 * i for i in range(111)]
    longitudes = [99.0 + 0.2 * j for j in range(141)]
    matrix = np.array(
        [[1 if is_land(lon, lat) else 0 for lon in longitudes]
         for lat in latitudes],
        dtype=np.uint8,
    )

    assert matrix.shape == (111, 141)
    assert matrix.size == 15651
    assert np.isin(matrix, (0, 1)).all()

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Array row zero is north; white=land and black=water.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    assert image.size == (141, 111)
    assert set(image.getdata()).issubset({(0, 0, 0), (255, 255, 255)})
    image.save(out / "candidate.png")

    enlarged = image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST)
    assert enlarged.size == (1128, 888)
    enlarged.save(out / "candidate_8x.png")

    # Validate the emitted CSV shape and alphabet.
    with (out / "candidate.csv").open("r", newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))
    assert len(rows) == 111
    assert all(len(row) == 141 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)


if __name__ == "__main__":
    main()
