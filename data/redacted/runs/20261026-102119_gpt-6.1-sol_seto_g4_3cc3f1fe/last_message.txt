我会按“手写近似多边形 + 点在多边形内”的规则生成判定，不会把规则计算说成逐点推理。由于当前交互要求最终直接给脚本，我将把所有输出生成、尺寸检查和二值校验都放进脚本中。
下面脚本仅使用手写的近似海岸线、多边形和椭圆岛屿规则，通过点在多边形内判断每个采样点。不会联网、读取地图或调用地理数据库。

def polygon(points):
    return tuple((float(x), float(y)) for x, y in points)


def ellipse(cx, cy, rx, ry, n=24, phase=0.0):
    result = []
    for i in range(n):
        a = phase + 2.0 * np.pi * i / n
        result.append((cx + rx * np.cos(a), cy + ry * np.sin(a)))
    return polygon(result)


def point_in_polygon(x, y, poly):
    inside = False
    j = len(poly) - 1

    for i in range(len(poly)):
        xi, yi = poly[i]
        xj, yj = poly[j]

        crosses = ((yi > y) != (yj > y))
        if crosses:
            intersection_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < intersection_x:
                inside = not inside

        j = i

    return inside

def classify(lon, lat):
    for poly in LAND_POLYGONS:
        if point_in_polygon(lon, lat, poly):
            return 1
    return 0


def build_matrix():
    lats = [LAT_NORTH - i * LAT_STEP for i in range(ROWS)]
    lons = [LON_WEST + j * LON_STEP for j in range(COLS)]

    matrix = np.zeros((ROWS, COLS), dtype=np.uint8)

    for row, lat in enumerate(lats):
        for col, lon in enumerate(lons):
            matrix[row, col] = classify(lon, lat)

    return matrix


def write_csv(matrix, path):
    with path.open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        for row in matrix:
            writer.writerow(int(value) for value in row)


def write_images(matrix, output_dir):
    image_array = matrix * 255
    image = Image.fromarray(image_array.astype(np.uint8), mode="L")

    if image.size != (253, 64):
        raise AssertionError(f"unexpected raw image size: {image.size}")

    image.save(output_dir / "candidate.png")

    nearest = getattr(Image, "Resampling", Image).NEAREST
    enlarged = image.resize((2024, 512), resample=nearest)

    if enlarged.size != (2024, 512):
        raise AssertionError(f"unexpected enlarged image size: {enlarged.size}")

    enlarged.save(output_dir / "candidate_8x.png")


def validate(matrix, csv_path, raw_png, enlarged_png):
    if matrix.shape != (64, 253):
        raise AssertionError(f"unexpected matrix shape: {matrix.shape}")

    if matrix.size != 16192:
        raise AssertionError(f"unexpected value count: {matrix.size}")

    if not np.all((matrix == 0) | (matrix == 1)):
        raise AssertionError("matrix contains values other than 0 and 1")

    with csv_path.open("r", encoding="ascii", newline="") as f:
        rows = list(csv.reader(f))

    if len(rows) != 64:
        raise AssertionError("CSV does not contain 64 rows")

    for row in rows:
        if len(row) != 253:
            raise AssertionError("CSV row does not contain 253 values")
        if any(value not in ("0", "1") for value in row):
            raise AssertionError("CSV contains a non-binary value")

    raw = Image.open(raw_png)
    enlarged = Image.open(enlarged_png)

    if raw.size != (253, 64):
        raise AssertionError("raw PNG has the wrong dimensions")
    if enlarged.size != (2024, 512):
        raise AssertionError("enlarged PNG has the wrong dimensions")

    raw_values = set(raw.convert("L").getdata())
    enlarged_values = set(enlarged.convert("L").getdata())

    if not raw_values.issubset({0, 255}):
        raise AssertionError("raw PNG is not strictly black and white")
    if not enlarged_values.issubset({0, 255}):
        raise AssertionError("enlarged PNG is not strictly black and white")

    expected = np.asarray(raw.convert("L"), dtype=np.uint8)
    expected = expected.reshape((64, 253))
    if not np.array_equal(expected, matrix * 255):
        raise AssertionError("raw PNG does not match the matrix")

    expected_enlarged = np.repeat(np.repeat(expected, 8, axis=0), 8, axis=1)
    actual_enlarged = np.asarray(enlarged.convert("L"), dtype=np.uint8)

    if not np.array_equal(actual_enlarged, expected_enlarged):
        raise AssertionError("enlarged PNG is not an exact nearest-neighbor enlargement")


def main():
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()

    csv_path = output_dir / "candidate.csv"
    raw_png = output_dir / "candidate.png"
    enlarged_png = output_dir / "candidate_8x.png"

    write_csv(matrix, csv_path)
    write_images(matrix, output_dir)
    validate(matrix, csv_path, raw_png, enlarged_png)

    print(f"matrix: {matrix.shape[0]} x {matrix.shape[1]} = {matrix.size}")
    print(f"land points: {int(matrix.sum())}")
    print(f"water points: {int(matrix.size - matrix.sum())}")
    print(f"CSV: {csv_path}")
    print(f"raw PNG: {raw_png} ({Image.open(raw_png).size[0]}x{Image.open(raw_png).size[1]})")
    print(
        f"8x PNG: {enlarged_png} "
        f"({Image.open(enlarged_png).size[0]}x{Image.open(enlarged_png).size[1]})"
    )
    print("validation: passed")


if __name__ == "__main__":
    main()
