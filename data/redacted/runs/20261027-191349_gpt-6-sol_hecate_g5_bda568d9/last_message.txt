以下脚本用手工近似的经纬度多边形表示大陆、海岸岛屿和较大的近岸岛群，再以点在多边形内判定每个采样点；不是逐点独立推理，也未查询外部地图数据。当前环境按要求未调用工具，因此我无法在此生成或展示运行后的图片与文件。保存并运行脚本后，它会在当前目录的 `outputs/` 中生成三项结果。

依赖：Python 标准库、NumPy、Pillow。

```python
#!/usr/bin/env python3
"""Generate an approximate Hecate Strait land/water raster from hand-drawn polygons."""

from pathlib import Path

import numpy as np
from PIL import Image

N = 127
LAT_NORTH, LAT_SOUTH = 55.0, 53.0
LON_WEST, LON_EAST = -131.0, -129.0

# Hand-approximated land outlines, in (longitude, latitude).
# Each polygon is closed by the point-in-polygon routine.
POLYGONS = [
    # North-coast mainland, including the coast and the eastern frame edge.
    [
        (-130.33, 55.00), (-130.18, 54.87), (-130.12, 54.75),
        (-130.23, 54.65), (-130.10, 54.55), (-130.20, 54.43),
        (-130.05, 54.33), (-130.16, 54.22), (-130.02, 54.10),
        (-130.14, 53.98), (-129.98, 53.87), (-130.06, 53.76),
        (-129.91, 53.65), (-129.99, 53.53), (-129.85, 53.41),
        (-129.91, 53.28), (-129.76, 53.16), (-129.79, 53.00),
        (-129.00, 53.00), (-129.00, 55.00),
    ],
    # A very narrow eastern fringe of Graham Island at the western edge.
    [
        (-131.00, 55.00), (-130.94, 55.00), (-130.93, 54.94),
        (-130.95, 54.88), (-130.92, 54.81), (-130.94, 54.75),
        (-130.91, 54.69), (-130.94, 54.62), (-130.91, 54.56),
        (-130.94, 54.50), (-130.91, 54.44), (-130.95, 54.38),
        (-130.93, 54.31), (-130.96, 54.24), (-130.94, 54.17),
        (-130.98, 54.10), (-131.00, 54.06),
    ],
    # Eastern coastal fragments of Moresby Island.
    [
        (-131.00, 53.89), (-130.96, 53.85), (-130.94, 53.79),
        (-130.96, 53.72), (-130.93, 53.67), (-130.96, 53.61),
        (-130.93, 53.54), (-130.97, 53.49), (-131.00, 53.46),
    ],
    # Banks Island.
    [
        (-130.18, 53.98), (-130.08, 53.96), (-129.98, 53.89),
        (-129.91, 53.78), (-129.88, 53.64), (-129.90, 53.48),
        (-129.96, 53.35), (-130.05, 53.25), (-130.15, 53.22),
        (-130.21, 53.30), (-130.23, 53.43), (-130.20, 53.57),
        (-130.23, 53.72), (-130.22, 53.87),
    ],
    # Pitt Island, clipped by the southern edge of the sampling frame.
    [
        (-130.55, 53.49), (-130.45, 53.46), (-130.37, 53.38),
        (-130.31, 53.27), (-130.28, 53.12), (-130.30, 53.00),
        (-130.63, 53.00), (-130.65, 53.14), (-130.62, 53.30),
        (-130.59, 53.42),
    ],
    # Dundas Island and small nearby islands.
    [
        (-130.83, 54.74), (-130.77, 54.72), (-130.72, 54.67),
        (-130.73, 54.61), (-130.79, 54.58), (-130.85, 54.61),
        (-130.87, 54.68),
    ],
    [
        (-130.69, 54.78), (-130.66, 54.77), (-130.64, 54.74),
        (-130.67, 54.71), (-130.70, 54.73),
    ],
    # Small nearshore islands and reefs represented as land only where sampled.
    [
        (-130.64, 54.48), (-130.61, 54.46), (-130.58, 54.43),
        (-130.60, 54.40), (-130.64, 54.42),
    ],
    [
        (-130.49, 54.12), (-130.45, 54.11), (-130.42, 54.08),
        (-130.44, 54.05), (-130.48, 54.07),
    ],
    [
        (-130.47, 53.83), (-130.43, 53.81), (-130.41, 53.77),
        (-130.44, 53.74), (-130.48, 53.77),
    ],
]


def contains(point_lon, point_lat, polygon):
    """Return whether a point is inside a polygon using ray crossing."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        crosses = (yi > point_lat) != (yj > point_lat)
        if crosses:
            crossing_lon = (xj - xi) * (point_lat - yi) / (yj - yi) + xi
            if point_lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    latitudes = np.linspace(LAT_NORTH, LAT_SOUTH, N)
    longitudes = np.linspace(LON_WEST, LON_EAST, N)

    # 1 = land, 0 = water. Every pixel is classified at its coordinate point.
    matrix = np.zeros((N, N), dtype=np.uint8)
    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = any(
                contains(float(lon), float(lat), polygon)
                for polygon in POLYGONS
            )

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Land is white; water is black. Matrix row order is north to south.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(out / "candidate.png")
    image.resize((N * 8, N * 8), resample=Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )


if __name__ == "__main__":
    main()
```
