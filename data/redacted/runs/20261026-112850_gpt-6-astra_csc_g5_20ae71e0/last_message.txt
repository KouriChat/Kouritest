本次无法在对话中实际运行脚本或附上生成后的图片；运行脚本后，文件会写入 `outputs/`。依赖仅为 Python 标准库、NumPy 和 Pillow。

def point_on_segment(px, py, ax, ay, bx, by):
    cross = (px - ax) * (by - ay) - (py - ay) * (bx - ax)
    if abs(cross) > 1e-10:
        return False
    return min(ax, bx) - 1e-10 <= px <= max(ax, bx) + 1e-10 and \
           min(ay, by) - 1e-10 <= py <= max(ay, by) + 1e-10

def inside_polygon(lon, lat, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if point_on_segment(lon, lat, xi, yi, xj, yj):
            return True
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside

def is_land(lon, lat):
    if any(inside_polygon(lon, lat, polygon) for polygon in LAND_POLYGONS):
        return True
    # Longitude distance is scaled by latitude for approximately round islets.
    cos_lat = max(0.1, math.cos(math.radians(lat)))
    for center_lon, center_lat, radius in ISLANDS:
        dx = (lon - center_lon) * cos_lat
        dy = lat - center_lat
        if dx * dx + dy * dy <= radius * radius:
            return True
    return False

def main():
    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    latitudes = [23.0 - 0.2 * row for row in range(111)]
    longitudes = [99.0 + 0.2 * col for col in range(141)]
    matrix = np.zeros((111, 141), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(is_land(lon, lat))

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Image convention: land white, water black.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(out_dir / "candidate.png")
    image.resize((1128, 888), resample=Image.Resampling.NEAREST).save(
        out_dir / "candidate_8x.png"
