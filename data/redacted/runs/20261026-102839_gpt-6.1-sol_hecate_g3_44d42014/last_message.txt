以下脚本用手写的近似海岸线和岛屿多边形，对每个采样点执行点在多边形判断；并非逐点独立推理。仅使用 Python 标准库、NumPy 和 Pillow。此环境不能运行脚本或展示生成的 PNG；运行脚本会在当前目录的 `outputs/` 下生成并校验三份文件。

def point_in_polygon(lon, lat, polygon):
    """Ray-crossing point test; polygon edges are handled consistently."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        # A tiny denominator guard covers horizontal edges.
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    if point_in_polygon(lon, lat, MAINLAND):
        return 1
    return int(any(point_in_polygon(lon, lat, island) for island in ISLANDS))


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    # North to south; west to east. The final coordinates are exactly 53N, 129W.
    latitudes = [55.0 - i * STEP for i in range(N)]
    longitudes = [-131.0 + i * STEP for i in range(N)]

    matrix = np.zeros((N, N), dtype=np.uint8)
    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = is_land(lon, lat)

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as handle:
        writer = csv.writer(handle, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # 1 -> white land; 0 -> black water.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    image.resize((N * 8, N * 8), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    # Verify the requested dimensions, values, and nearest-neighbor expansion.
    with csv_path.open("r", newline="", encoding="ascii") as handle:
        rows = list(csv.reader(handle))
    assert len(rows) == N and all(len(row) == N for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(len(row) for row in rows) == 16129

    original = Image.open(output_dir / "candidate.png").convert("RGB")
    enlarged = Image.open(output_dir / "candidate_8x.png").convert("RGB")
    assert original.size == (127, 127)
    assert set(original.getdata()) <= {(0, 0, 0), (255, 255, 255)}
    assert enlarged.size == (1016, 1016)
    assert enlarged == original.resize((1016, 1016), Image.Resampling.NEAREST)


if __name__ == "__main__":
    main()
