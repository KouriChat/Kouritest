def point_in_polygon(lon, lat, polygon):
    """Ray-casting point-in-polygon test with boundary inclusion."""
    inside = False
    n = len(polygon)
    eps = 1e-12

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) <= eps:
            if (
                min(x1, x2) - eps <= lon <= max(x1, x2) + eps
                and min(y1, y2) - eps <= lat <= max(y1, y2) + eps
            ):
                return True

        crosses = ((y1 > lat) != (y2 > lat))
        if crosses:
            intersection_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < intersection_lon:
                inside = not inside

    return inside


def point_in_ellipse(lon, lat, cx, cy, rx, ry):
    dx = (lon - cx) / rx
    dy = (lat - cy) / ry
    return dx * dx + dy * dy <= 1.0


def is_land(lon, lat):
    for polygon in POLYGONS:
        if point_in_polygon(lon, lat, polygon):
            return 1

    for cx, cy, rx, ry in ELLIPSES:
        if point_in_ellipse(lon, lat, cx, cy, rx, ry):
            return 1

    return 0


def build_matrix():
    latitudes = np.array(
        [23.0 - 0.2 * i for i in range(111)],
        dtype=np.float64,
    )
    longitudes = np.array(
        [99.0 + 0.2 * i for i in range(141)],
        dtype=np.float64,
    )

    matrix = np.zeros((111, 141), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = is_land(float(lon), float(lat))

    return matrix


def main():
    matrix = build_matrix()

    assert matrix.shape == (111, 141)
    assert matrix.size == 15651
    assert np.all((matrix == 0) | (matrix == 1))

    csv_path = os.path.join(OUT_DIR, "candidate.csv")
    png_path = os.path.join(OUT_DIR, "candidate.png")
    png_8x_path = os.path.join(OUT_DIR, "candidate_8x.png")

    np.savetxt(
        csv_path,
        matrix,
        fmt="%d",
        delimiter=",",
    )

if __name__ == "__main__":
    main()
