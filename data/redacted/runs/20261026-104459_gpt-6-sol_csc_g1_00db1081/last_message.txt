def point_in_polygon(lon, lat, polygon):
    """Ray-casting point test for a hand-written polygon."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    for polygon in LAND_POLYGONS:
        if point_in_polygon(lon, lat, polygon):
            return True
    for center_lon, center_lat, radius_lon, radius_lat in ISLANDS:
        dx = (lon - center_lon) / radius_lon
        dy = (lat - center_lat) / radius_lat
        if dx * dx + dy * dy <= 1.0:
            return True
    return False


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    latitudes = 23.0 - 0.2 * np.arange(111)
    longitudes = 99.0 + 0.2 * np.arange(141)
    matrix = np.zeros((111, 141), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(is_land(float(lon), float(lat)))

    np.savetxt(output_dir / "candidate.csv", matrix, fmt="%d", delimiter=",")

    # 0 is black water; 1 is white land.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    assert matrix.shape == (111, 141)
    assert int(matrix.size) == 15651
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)
    assert enlarged.size == (1128, 888)


if __name__ == "__main__":
    main()
