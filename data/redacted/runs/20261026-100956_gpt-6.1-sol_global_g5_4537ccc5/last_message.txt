def in_polygon(lon, lat, polygon):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    if any(in_polygon(lon, lat, lake) for lake in LAKES):
        return 0
    return int(any(
        in_polygon(lon, lat, polygon)
        for polygon in CONTINENTS + ISLANDS
    ))


latitudes = np.arange(89, -90, -2, dtype=int)
