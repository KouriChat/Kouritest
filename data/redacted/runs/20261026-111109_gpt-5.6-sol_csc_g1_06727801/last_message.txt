def point_in_polygon(x, y, polygon):
    """Ray-casting point test; boundary points count as land."""
    inside = False
    n = len(polygon)
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]
        dx, dy = x2 - x1, y2 - y1
        cross = (x - x1) * dy - (y - y1) * dx
        if abs(cross) < 1e-10 and min(x1, x2) - 1e-10 <= x <= max(x1, x2) + 1e-10:
            if min(y1, y2) - 1e-10 <= y <= max(y1, y2) + 1e-10:
                return True
        if (y1 > y) != (y2 > y):
            at_x = x1 + (y - y1) * dx / dy
            if x < at_x:
                inside = not inside
    return inside


def circle_polygon(lon, lat, rx, ry, sides=12):
    """Small hand-placed island/reef approximation."""
    return [
        (lon + rx * math.cos(2 * math.pi * i / sides),
         lat + ry * math.sin(2 * math.pi * i / sides))
        for i in range(sides)
    ]

def main():
    longitudes = 99.0 + 0.2 * np.arange(141)
    latitudes = 23.0 - 0.2 * np.arange(111)
    matrix = np.zeros((111, 141), dtype=np.uint8)

    # Each sampled coordinate is classified independently by the polygons.
    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(any(
                point_in_polygon(float(lon), float(lat), polygon)
                for polygon in LAND
            ))

    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # 1 is white land; 0 is black water.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out / "candidate.png")
    enlarged = image.resize((1128, 888), resample=Image.Resampling.NEAREST)
    enlarged.save(out / "candidate_8x.png")

    assert image.size == (141, 111)

    assert set(image.getdata()) <= {0, 255}
