以下脚本使用手写的近似海岸线和岛屿多边形，再按每个采样点的位置进行点内多边形判断；不是对采样点逐一独立推理。运行后会在当前目录创建 `outputs/`，并生成 CSV、原始 PNG 和 8 倍最近邻 PNG。此处无法直接运行脚本或展示生成后的图片。

依赖：Python 标准库、NumPy、Pillow。

def point_on_segment(x, y, x1, y1, x2, y2, epsilon=1e-10):
    cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
    if abs(cross) > epsilon:
        return False
    return (
        min(x1, x2) - epsilon <= x <= max(x1, x2) + epsilon
        and min(y1, y2) - epsilon <= y <= max(y1, y2) + epsilon
    )


def point_in_polygon(x, y, polygon):
    inside = False
    for i, (x1, y1) in enumerate(polygon):
        x2, y2 = polygon[(i + 1) % len(polygon)]
        if point_on_segment(x, y, x1, y1, x2, y2):
            return True
        if (y1 > y) != (y2 > y):
            crossing_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing_x:
                inside = not inside
    return inside


def build_matrix():
    # Exact interval is 2/126 degrees, producing 127 samples on each axis.
    latitudes = np.linspace(55.0, 53.0, 127)
    longitudes = np.linspace(-131.0, -129.0, 127)
    matrix = np.zeros((127, 127), dtype=np.uint8)

    for row, latitude in enumerate(latitudes):
        for col, longitude in enumerate(longitudes):
            matrix[row, col] = int(
                any(
                    point_in_polygon(float(longitude), float(latitude), polygon)
                    for polygon in POLYGONS
                )
            )
    return matrix


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()

    with (output_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Pillow uses RGB here to make the required black/white pixel values explicit.
    pixels = np.where(matrix[:, :, None] == 1, 255, 0).astype(np.uint8)
    pixels = np.repeat(pixels, 3, axis=2)
    image = Image.fromarray(pixels, mode="RGB")
    image.save(output_dir / "candidate.png")
    image.resize((1016, 1016), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    assert matrix.shape == (127, 127)
    assert matrix.size == 16129
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (127, 127)
    assert set(image.getdata()).issubset({(0, 0, 0), (255, 255, 255)})
    assert Image.open(output_dir / "candidate_8x.png").size == (1016, 1016)


if __name__ == "__main__":
    main()
