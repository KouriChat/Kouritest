def contains(polygon, x, y):
    """Even-odd point-in-polygon test, with boundary counted as land."""
    inside = False
    previous = polygon[-1]
    for current in polygon:
        x1, y1 = previous
        x2, y2 = current
        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if (
            abs(cross) < 1e-10
            and min(x1, x2) - 1e-10 <= x <= max(x1, x2) + 1e-10
            and min(y1, y2) - 1e-10 <= y <= max(y1, y2) + 1e-10
        ):
            return True
        if (y1 > y) != (y2 > y):
            intersection_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < intersection_x:
                inside = not inside
        previous = current
    return inside

    # before their conversion to decimal degrees.
    matrix = [
        [
            int(any(contains(poly, (495 + col) / 5, (115 - row) / 5)
                    for poly in POLYGONS))
            for col in range(141)
        ]
        for row in range(111)
    ]

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as stream:
        csv.writer(stream, lineterminator="\n").writerows(matrix)

    image = Image.new("L", (141, 111))
    image.putdata([255 * value for row in matrix for value in row])
    image.save(output_dir / "candidate.png")

    enlarged = image.resize((1128, 888), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    # Check the written artifacts, not only the in-memory matrix.
    with csv_path.open(newline="", encoding="ascii") as stream:
        rows = list(csv.reader(stream))
    assert len(rows) == 111
    assert all(len(row) == 141 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(len(row) for row in rows) == 15651

    with Image.open(output_dir / "candidate.png") as original:
        original.load()
        assert original.size == (141, 111)
        assert set(original.getdata()) <= {0, 255}
        assert list(original.getdata()) == [
            255 * int(value) for row in rows for value in row
        ]

        with Image.open(output_dir / "candidate_8x.png") as large:
            large.load()
            assert large.size == (1128, 888)
            assert set(large.getdata()) <= {0, 255}
            expected = original.resize(
                large.size, resample=Image.Resampling.NEAREST
            )
            assert large.tobytes() == expected.tobytes()

    print("Validated: 111 x 141 = 15651 binary values.")
    print("Wrote outputs/candidate.csv")
    print("Wrote outputs/candidate.png")
    print("Wrote outputs/candidate_8x.png")


if __name__ == "__main__":
    main()
