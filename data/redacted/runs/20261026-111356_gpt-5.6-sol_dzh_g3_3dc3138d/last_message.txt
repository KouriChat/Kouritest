def contains(point, polygon):
    """Ray-casting point-in-polygon test."""
    x, y = point
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def main():
    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    # Rows are north to south; columns are west to east.
    latitudes = 45.0 - 0.2 * np.arange(71)
    longitudes = -9.0 + 0.2 * np.arange(231)
    matrix = np.zeros((71, 231), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(
                any(contains((float(lon), float(lat)), poly) for poly in POLYGONS)
            )

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    image.resize((231 * 8, 71 * 8), resample=Image.Resampling.NEAREST).save(

    assert matrix.size == 16401

    print("Wrote outputs/candidate.csv, candidate.png, candidate_8x.png")


if __name__ == "__main__":
