def point_in_polygon(lon, lat, ring):
    """Ray crossing test; boundary points are treated as land."""
    inside = False
    n = len(ring)
    for i in range(n):
        x1, y1 = ring[i]
        x2, y2 = ring[(i + 1) % n]
        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-10:
            if (min(x1, x2) - 1e-10 <= lon <= max(x1, x2) + 1e-10
                    and min(y1, y2) - 1e-10 <= lat <= max(y1, y2) + 1e-10):
                return True
        if (y1 > lat) != (y2 > lat):
            x_cross = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < x_cross:
                inside = not inside
    return inside


def is_land(lon, lat):
    if any(point_in_polygon(lon, lat, ring) for ring in POLYGONS):
        return True
    for center_lon, center_lat, radius in ISLANDS:
        # Scale longitude distance by latitude so radii are approximately circular.
        dx = (lon - center_lon) * math.cos(math.radians(lat))
        dy = lat - center_lat
        if dx * dx + dy * dy <= radius * radius:
            return True
    return False


def main():
    latitudes = [23.0 - 0.2 * i for i in range(111)]
    longitudes = [99.0 + 0.2 * j for j in range(141)]
    matrix = np.array(
        [[1 if is_land(lon, lat) else 0 for lon in longitudes]
         for lat in latitudes],
        dtype=np.uint8,
    )

    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as file:
        writer = csv.writer(file, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Matrix value 1 (land) -> white; value 0 (water) -> black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    image.resize((141 * 8, 111 * 8), Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)
    assert int(matrix.size) == 15651
    assert set(np.unique(matrix).tolist()) <= {0, 1}
    assert image.size == (141, 111)
    assert set(image.getdata()) <= {0, 255}
    assert Image.open(output_dir / "candidate_8x.png").size == (1128, 888)


if __name__ == "__main__":
    main()
