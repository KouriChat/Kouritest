def contains(lon, lat, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing:
                inside = not inside
        j = i
    return inside

def is_land(lon, lat):
    if any(contains(lon, lat, p) for p in
           (MAINLAND, MALAY_PENINSULA, BORNEO, SUMATRA)):
        return 1
    if any(contains(lon, lat, p) for p in ISLANDS):
        return 1
    for cx, cy, r in SMALL_FEATURES:
        # Compact circular approximations for small islands and reef rocks.
        if ((lon - cx) / r) ** 2 + ((lat - cy) / (r * 0.8)) ** 2 <= 1:
            return 1
    return 0

def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    lats = [23.0 - 0.2 * i for i in range(111)]
    lons = [99.0 + 0.2 * j for j in range(141)]
    matrix = np.array(
        [[is_land(lon, lat) for lon in lons] for lat in lats],
        dtype=np.uint8,
    )

    with (out / "candidate.csv").open("w", newline="", encoding="utf-8") as f:
        csv.writer(f, lineterminator="\n").writerows(matrix.tolist())

    # Matrix value 1 maps to white; value 0 maps to black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out / "candidate.png")
    image.resize((141 * 8, 111 * 8), Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)

if __name__ == "__main__":
    main()
