def inside_polygon(lon, lat, polygon):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    return int(any(inside_polygon(lon, lat, polygon) for polygon in LAND))


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    rows = []
    for row in range(111):
        lat = 23.0 - row * 0.2
        rows.append([is_land(99.0 + col * 0.2, lat) for col in range(141)])

    with (output_dir / "candidate.csv").open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(rows)

    matrix = np.asarray(rows, dtype=np.uint8)
    # Matrix 1 is land: white. Matrix 0 is water: black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"

if __name__ == "__main__":
