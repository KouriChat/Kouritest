以下脚本把手写的粗略海岸线和岛屿多边形作为规则，通过点在多边形内判断采样点；不是对每个点分别进行语言模型推理。未使用外部地理数据。当前对话环境无法执行脚本或直接附上生成后的 PNG；运行后会在 `outputs/` 中生成 CSV 和两张图片。

依赖：Python、NumPy、Pillow。

from pathlib import Path

def inside_ring(lon, lat, ring):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(ring) - 1
    for i, (xi, yi) in enumerate(ring):
        xj, yj = ring[j]
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def make_matrix():
    result = np.zeros((len(LATS), len(LONS)), dtype=np.uint8)
    for row, lat in enumerate(LATS):
        for col, lon in enumerate(LONS):
            if any(inside_ring(lon, lat, ring) for ring in LAND):
                result[row, col] = 1
            if any(inside_ring(lon, lat, ring) for ring in WATER):
                result[row, col] = 0
    return result


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    matrix = make_matrix()

    assert matrix.shape == (71, 231)
    assert matrix.size == 16401
    assert np.isin(matrix, (0, 1)).all()

    with (OUT / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # PIL mode "L": 0 is black water, 255 is white land.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(OUT / "candidate.png")
    enlarged = image.resize((1848, 568), resample=Image.Resampling.NEAREST)
    enlarged.save(OUT / "candidate_8x.png")

    # Validate CSV shape/content and exact nearest-neighbor enlargement.
    with (OUT / "candidate.csv").open(newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))
    assert len(rows) == 71 and all(len(row) == 231 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert image.size == (231, 71)
    assert set(image.getdata()) <= {0, 255}
    assert enlarged.size == (1848, 568)
    assert enlarged.resize((231, 71), Image.Resampling.NEAREST).tobytes() == image.tobytes()


if __name__ == "__main__":
    main()
