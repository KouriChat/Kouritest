def inside_polygon(x, y, points):
    """射线法；仅判断手写多边形，不读取任何地理数据。"""
    hit = False
    j = len(points) - 1
    for i, (xi, yi) in enumerate(points):
        xj, yj = points[j]
        if (yi > y) != (yj > y):
            boundary_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < boundary_x:
                hit = not hit
        j = i
    return hit


def antarctica(x, y):
    """用手写的沿经度变化的近似海岸纬度判断南极大陆。"""
    coast = [
        (-180, -72), (-165, -76), (-150, -78),
        (-135, -77), (-120, -73), (-100, -72),
        (-80, -71), (-72, -70), (-69, -67),
        (-64, -64), (-60, -63), (-56, -65),
        (-62, -70), (-50, -75), (-30, -76),
        (-15, -73), (0, -70), (20, -69),
        (40, -68), (60, -66), (80, -67),
        (100, -66), (120, -66), (140, -67),
        (160, -69), (180, -72),
    ]
    for (x0, y0), (x1, y1) in zip(coast, coast[1:]):
        if x0 <= x <= x1:
            latitude = y0 + (y1 - y0) * (x - x0) / (x1 - x0)
            return y <= latitude
    return False


def land_at(lon, lat):
    if antarctica(lon, lat):
        return 1
    if any(inside_polygon(lon, lat, p) for p in LAND_POLYGONS):
        return 1
    return 0


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    latitudes = range(89, -90, -2)
    longitudes = range(-179, 180, 2)
    matrix = np.array(
        [[land_at(lon, lat) for lon in longitudes]
         for lat in latitudes],
        dtype=np.uint8,
    )

    assert matrix.shape == (90, 180)
    assert matrix.size == 16200
    assert set(np.unique(matrix).tolist()).issubset({0, 1})

    csv_path = out / "candidate.csv"
    with csv_path.open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f, lineterminator="\n")

    # 直接将每个矩阵值映射成一个黑色或白色像素。
    original = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    enlarged = original.resize((1440, 720), Image.Resampling.NEAREST)
    original.save(out / "candidate.png")
    enlarged.save(out / "candidate_8x.png")

    # 检查写出的文件，而不只检查内存中的数组。
    with csv_path.open(newline="", encoding="utf-8") as f:

    assert len(rows) == 90
    assert all(len(row) == 180 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert np.array_equal(
        np.array(rows, dtype=np.uint8), matrix
    )

    with Image.open(out / "candidate.png") as img:
        assert img.size == (180, 90)
        assert set(img.convert("RGB").getdata()) <= {

    with Image.open(out / "candidate_8x.png") as img:

        assert set(img.convert("RGB").getdata()) <= {

        assert np.array_equal(

    for name in ("candidate.csv", "candidate.png", "candidate_8x.png"):
