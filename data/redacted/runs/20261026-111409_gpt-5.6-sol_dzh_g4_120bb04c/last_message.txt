def contains(lon, lat, ring):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(ring) - 1
    for i, (xi, yi) in enumerate(ring):
        xj, yj = ring[j]
        if (yi > lat) != (yj > lat):
            crossing_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_x:
                inside = not inside
        j = i
    return inside


def main():

    lons = -9.0 + 0.2 * np.arange(231)
    matrix = np.zeros((71, 231), dtype=np.uint8)

    for row, lat in enumerate(lats):
        for col, lon in enumerate(lons):
            matrix[row, col] = int(
                any(contains(float(lon), float(lat), ring) for ring in LAND)
            )

    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)
