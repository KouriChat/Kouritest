下面脚本仅使用手写近似多边形规则判断采样点，不联网、不读取外部地图或地理数据库。规则先判断点是否落入任一陆地多边形，再用手写水道多边形扣除海峡和海湾。

运行依赖：`Python 3`、`numpy`、`Pillow`。

def compile_polygons(polygons):
    compiled = []
    for points in polygons:
        xs = [p[0] for p in points]
        ys = [p[1] for p in points]
        compiled.append((
            min(xs), max(xs), min(ys), max(ys), points
        ))
    return compiled


LAND = compile_polygons(LAND_POLYGONS)
WATER = compile_polygons(WATER_CUTS)


def point_in_polygon(lon, lat, polygon):
    min_x, max_x, min_y, max_y, points = polygon

    if lon < min_x or lon > max_x or lat < min_y or lat > max_y:
        return False

    inside = False
    j = len(points) - 1

    for i in range(len(points)):
        xi, yi = points[i]
        xj, yj = points[j]

        intersects = ((yi > lat) != (yj > lat))
        if intersects:
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside

        j = i

    return inside


def is_land(lon, lat):
    in_land = any(point_in_polygon(lon, lat, shape) for shape in LAND)
    if not in_land:
        return False

    in_water_cut = any(point_in_polygon(lon, lat, shape) for shape in WATER)
    return not in_water_cut


def build_matrix():
    result = np.zeros((N_ROWS, N_COLS), dtype=np.uint8)

    for row in range(N_ROWS):
        lat = LAT_NORTH - row * STEP
        for col in range(N_COLS):
            lon = LON_WEST + col * STEP
            result[row, col] = 1 if is_land(lon, lat) else 0

    return result


def write_outputs(matrix):
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    png_path = output_dir / "candidate.png"
    large_png_path = output_dir / "candidate_8x.png"

    with csv_path.open("w", newline="", encoding="ascii") as handle:
        writer = csv.writer(handle, lineterminator="\n")
        for row in matrix:
            writer.writerow([int(value) for value in row])

    # 0 -> black water, 1 -> white land.
    image_array = matrix * 255
    image = Image.fromarray(image_array, mode="L")
    image.save(png_path)

    enlarged = image.resize(
        (N_COLS * 8, N_ROWS * 8),
        resample=Image.Resampling.NEAREST
    )
    enlarged.save(large_png_path)

    return csv_path, png_path, large_png_path


def validate_outputs(matrix, csv_path, png_path, large_png_path):
    assert matrix.shape == (61, 241)
    assert matrix.size == 14701
    assert set(np.unique(matrix)).issubset({0, 1})

    with csv_path.open("r", newline="", encoding="ascii") as handle:
        rows = list(csv.reader(handle))

    assert len(rows) == 61
    assert all(len(row) == 241 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert all(value != "" for row in rows for value in row)

    with Image.open(png_path) as image:
        assert image.size == (241, 61)
        source = np.asarray(image.convert("L"))
        assert set(np.unique(source)).issubset({0, 255})
        assert np.array_equal(source, matrix * 255)

    with Image.open(large_png_path) as image:
        assert image.size == (1928, 488)
        enlarged = np.asarray(image.convert("L"))
        assert set(np.unique(enlarged)).issubset({0, 255})
        assert np.array_equal(enlarged[::8, ::8], matrix * 255)

        for row in range(61):
            for col in range(241):
                block = enlarged[
                    row * 8:(row + 1) * 8,
                    col * 8:(col + 1) * 8
                ]
                assert np.all(block == matrix[row, col] * 255)


def main():
    matrix = build_matrix()
    csv_path, png_path, large_png_path = write_outputs(matrix)
    validate_outputs(matrix, csv_path, png_path, large_png_path)

    print("matrix:", matrix.shape, "values:", matrix.size)
    print("csv:", csv_path)
    print("png:", png_path)
    print("png_8x:", large_png_path)


if __name__ == "__main__":
    main()
