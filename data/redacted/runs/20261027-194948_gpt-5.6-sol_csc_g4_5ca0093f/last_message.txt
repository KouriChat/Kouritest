def on_land(lon, lat):
    for polygon in LAND:
        inside = False
        j = len(polygon) - 1
        for i, (xi, yi) in enumerate(polygon):
            xj, yj = polygon[j]
            if (yi > lat) != (yj > lat):
                cross_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
                if lon < cross_x:
                    inside = not inside
            j = i
        if inside:
            return True
    return False

def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    rows = 111
    cols = 141
    matrix = np.zeros((rows, cols), dtype=np.uint8)

    for r in range(rows):
        lat = 23.0 - 0.2 * r
        for c in range(cols):
            lon = 99.0 + 0.2 * c
            matrix[r, c] = int(on_land(lon, lat))

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    image = Image.fromarray(np.where(matrix == 1, 255, 0).astype(np.uint8), mode="L")
    image.save(out / "candidate.png")
    image.resize((cols * 8, rows * 8), resample=Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)
    assert int(matrix.size) == 15651
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)
    assert set(image.getdata()).issubset({0, 255})
    assert Image.open(out / "candidate_8x.png").size == (1128, 888)

if __name__ == "__main__":
    main()
