def polygon_mask(vertices):
    """Point-in-polygon mask, with points on polygon edges included."""
    vertices = np.asarray(vertices, dtype=float)
    inside = np.zeros(X.shape, dtype=bool)
    boundary = np.zeros(X.shape, dtype=bool)
    tolerance = 1e-10

    xj, yj = vertices[-1]
    for xi, yi in vertices:
        dx = xi - xj
        dy = yi - yj

        cross = (X - xj) * dy - (Y - yj) * dx
        within = (
            (X >= min(xi, xj) - tolerance)
            & (X <= max(xi, xj) + tolerance)
            & (Y >= min(yi, yj) - tolerance)
            & (Y <= max(yi, yj) + tolerance)
        )
        boundary |= within & (np.abs(cross) <= tolerance)

        if abs(dy) > tolerance:
            intersects = ((yi > Y) != (yj > Y)) & (
                X < xj + (Y - yj) * dx / dy
            )
            inside ^= intersects

        xj, yj = xi, yi

    return inside | boundary


def ellipse_mask(cx, cy, rx, ry, angle_degrees=0.0):
    """Mask for a rotated ellipse in longitude/latitude coordinates."""
    angle = np.deg2rad(angle_degrees)
    ca = np.cos(angle)
    sa = np.sin(angle)
    dx = X - cx
    dy = Y - cy
    xr = ca * dx + sa * dy
    yr = -sa * dx + ca * dy
    return (xr / rx) ** 2 + (yr / ry) ** 2 <= 1.0
