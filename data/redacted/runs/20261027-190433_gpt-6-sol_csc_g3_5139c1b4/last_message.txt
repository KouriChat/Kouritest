def point_in_polygon(lon, lat, polygon):
    """Ray casting test; boundary points count as land."""
    inside = False
    n = len(polygon)
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        dx, dy = x2 - x1, y2 - y1
        cross = (lon - x1) * dy - (lat - y1) * dx
        if abs(cross) < 1e-10:
            if (min(x1, x2) - 1e-10 <= lon <= max(x1, x2) + 1e-10
                    and min(y1, y2) - 1e-10 <= lat <= max(y1, y2) + 1e-10):
                return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * dx / dy
            if lon < crossing_lon:
                inside = not inside
    return inside


def ellipse_contains(lon, lat, center_lon, center_lat, rx, ry):
    return ((lon - center_lon) / rx) ** 2 + ((lat - center_lat) / ry) ** 2 <= 1.0

def is_land(lon, lat):
    if point_in_polygon(lon, lat, MAINLAND):
        return True
    if point_in_polygon(lon, lat, INDOCHINA):
        return True
    if any(point_in_polygon(lon, lat, polygon) for polygon in ISLANDS):
        return True
    if any(point_in_polygon(lon, lat, polygon) for polygon in PHILIPPINES):
        return True
    return any(ellipse_contains(lon, lat, *item) for item in ISLAND_ELLIPSES)

            values.append(1 if is_land(lon, lat) else 0)
        rows.append(values)

    matrix = np.asarray(rows, dtype=np.uint8)
    if matrix.shape != (111, 141):

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:

    print("Generated 111 x 141 matrix and both PNG images in outputs/.")
