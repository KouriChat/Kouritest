def contains(lon, lat, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            cross = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < cross:
                inside = not inside
        j = i
    return inside


def main():
    latitudes = np.arange(89, -90, -2)
    longitudes = np.arange(-179, 180, 2)
    matrix = np.zeros((90, 180), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(any(contains(lon, lat, p) for p in POLYGONS))

    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)
    np.savetxt(output / "candidate.csv", matrix, fmt="%d", delimiter=",")

    pixels = np.where(matrix == 1, 255, 0).astype(np.uint8)
    image = Image.fromarray(pixels, mode="L")
    image.save(output / "candidate.png")
    image.resize((1440, 720), resample=Image.Resampling.NEAREST).save(
        output / "candidate_8x.png"
    )


if __name__ == "__main__":
    main()
