def points_in_polygon(lon_w, lat, polygon):
    """Vectorized ray-crossing test; coordinates are W longitude and N latitude."""
    inside = np.zeros(lon_w.shape, dtype=bool)
    x0, y0 = polygon[-1]
    for x1, y1 in polygon:
        crosses = (y0 > lat) != (y1 > lat)
        denominator = y1 - y0
        if denominator != 0:
            edge_x = (x1 - x0) * (lat - y0) / denominator + x0
            inside ^= crosses & (lon_w < edge_x)
        x0, y0 = x1, y1
    return inside


def main():
    latitudes = 79.0 - np.arange(61) * 0.1
    longitudes_w = 105.0 - np.arange(241) * 0.1
    lon_grid, lat_grid = np.meshgrid(longitudes_w, latitudes)

    land = np.zeros((61, 241), dtype=bool)
    for polygon in LAND:
        land |= points_in_polygon(lon_grid, lat_grid, polygon)

    matrix = land.astype(np.uint8)
    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    with (out_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Pixel value 255 is white land; 0 is black water.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out_dir / "candidate.png")
    image.resize((241 * 8, 61 * 8), Image.Resampling.NEAREST).save(

if __name__ == "__main__":
    main()
