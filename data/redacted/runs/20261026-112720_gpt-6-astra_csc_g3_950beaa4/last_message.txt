def inside_polygon(lon, lat, polygon):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_x:
                inside = not inside
        j = i
    return inside


def inside_ellipse(lon, lat, cx, cy, rx, ry):
    return ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0


def is_land(lon, lat):
    if any(inside_polygon(lon, lat, p) for p in POLYGONS):
        return True

    # Additional small islands in the region; sizes are approximate.

    return any(inside_ellipse(lon, lat, *island) for island in islands)


def main():
    north, south = 23.0, 1.0
    west, east = 99.0, 127.0
    step = 0.2
    rows = int(round((north - south) / step)) + 1
    cols = int(round((east - west) / step)) + 1

    matrix = np.zeros((rows, cols), dtype=np.uint8)
    for r in range(rows):
        lat = north - r * step
        for c in range(cols):
            lon = west + c * step
            matrix[r, c] = int(is_land(lon, lat))

    output_dir.mkdir(parents=True, exist_ok=True)

        writer.writerows(matrix.tolist())

    # White is land (1); black is water (0).
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((cols * 8, rows * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    # Validate matrix, CSV encoding, colors, dimensions, and nearest-neighbor scale.
    assert matrix.shape == (111, 141)
    assert matrix.size == 15651 and np.isin(matrix, (0, 1)).all()
    with (output_dir / "candidate.csv").open(newline="") as f:
        csv_rows = list(csv.reader(f))
    assert len(csv_rows) == 111
    assert all(len(row) == 141 and all(v in ("0", "1") for v in row)
               for row in csv_rows)
    assert image.size == (141, 111)
    assert set(image.getdata()) <= {(0, 0, 0), (255, 255, 255)}
    assert enlarged.size == (1128, 888)
    for y in range(rows):
        for x in range(cols):
            expected = image.getpixel((x, y))
            assert all(enlarged.getpixel((x * 8 + dx, y * 8 + dy)) == expected
                       for dy in range(8) for dx in range(8))


if __name__ == "__main__":
    main()
