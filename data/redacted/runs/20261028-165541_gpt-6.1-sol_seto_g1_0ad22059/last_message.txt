def point_in_polygon(x, y, polygon):
    inside = False
    n = len(polygon)
    j = n - 1
    for i in range(n):
        xi, yi = polygon[i]
        xj, yj = polygon[j]
        if (yi > y) != (yj > y):
            cross_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < cross_x:
                inside = not inside
        j = i
    return inside

def in_island(x, y, island):
    cx, cy, rx, ry, angle = island
    a = math.radians(angle)
    dx = x - cx
    dy = y - cy
    u = dx * math.cos(a) + dy * math.sin(a)
    v = -dx * math.sin(a) + dy * math.cos(a)
    return (u / rx) ** 2 + (v / ry) ** 2 <= 1.0

def is_land(x, y):
    land = any(point_in_polygon(x, y, p) for p in LAND_POLYGONS)
    if land and any(point_in_polygon(x, y, p) for p in WATER_POLYGONS):
        land = False
    if land:
        return True
    return any(in_island(x, y, island) for island in ISLANDS)

def main():
    rows, cols = 64, 253
    latitudes = [35.0 - i * (2.0 / 63.0) for i in range(rows)]
    longitudes = [129.0 + j * (8.0 / 252.0) for j in range(cols)]
    matrix = np.zeros((rows, cols), dtype=np.uint8)

    for r, lat in enumerate(latitudes):
        for c, lon in enumerate(longitudes):
            matrix[r, c] = 1 if is_land(lon, lat) else 0

    out_dir.mkdir(parents=True, exist_ok=True)

        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    enlarged = image.resize((cols * 8, rows * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(out_dir / "candidate_8x.png")
