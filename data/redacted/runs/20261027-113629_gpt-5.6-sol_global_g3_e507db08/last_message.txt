def point_in_polygon(x, y, polygon):
    """Odd-even ray casting; boundary points are treated as inside."""
    inside = False
    n = len(polygon)

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        # Boundary test.
        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if abs(cross) < 1e-10:
            if min(x1, x2) <= x <= max(x1, x2) and \
               min(y1, y2) <= y <= max(y1, y2):
                return True

        if (y1 > y) != (y2 > y):
            intersection_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if intersection_x >= x:
                inside = not inside

    return inside


def point_in_ellipse(lon, lat, ellipse):
    cx, cy, rx, ry = ellipse
    return ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0


# Approximate grounded Antarctic coastline. Ice shelves and surrounding sea
# remain water. Values are approximate northern coastline latitudes.
ANTARCTIC_EDGE = [
    (-180, -72),
    (-160, -74),
    (-140, -75),
    (-120, -74),
    (-100, -73),
    (-80, -72),
    (-68, -67),
    (-62, -63),  # Antarctic Peninsula
    (-56, -64),
    (-45, -70),
    (-30, -73),
    (0, -71),
    (30, -70),
    (60, -67),
    (90, -66),
    (120, -67),
    (150, -70),
    (180, -72),
]


def antarctic_coast_latitude(lon):
    for (x1, y1), (x2, y2) in zip(
        ANTARCTIC_EDGE[:-1], ANTARCTIC_EDGE[1:]
    ):
        if x1 <= lon <= x2:
            fraction = (lon - x1) / (x2 - x1)
            return y1 + fraction * (y2 - y1)

    raise ValueError("Longitude outside Antarctic interpolation range")


def is_land(lon, lat):
    # Grounded Antarctic land/ice.
    if lat <= antarctic_coast_latitude(lon):
        return True

    land = any(
        point_in_polygon(lon, lat, polygon)
        for polygon in LAND_POLYGONS
    )

    if not land:
        land = any(
            point_in_ellipse(lon, lat, ellipse)
            for ellipse in LAND_ELLIPSES
        )

    if not land:
        return False

    # Remove only ocean-connected waters. Inland lakes are deliberately not
    # removed because the requested classification counts them as land.
    if any(
        point_in_polygon(lon, lat, polygon)
        for polygon in WATER_POLYGONS
    ):
        return False

    return True


def build_matrix():
    latitudes = list(range(89, -90, -2))
    longitudes = list(range(-179, 180, 2))

    assert len(latitudes) == 90
    assert len(longitudes) == 180

    matrix = np.zeros((90, 180), dtype=np.uint8)

    for row, latitude in enumerate(latitudes):
        for column, longitude in enumerate(longitudes):
            matrix[row, column] = int(is_land(longitude, latitude))

    return matrix


def save_outputs(matrix):
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    original_path = output_dir / "candidate.png"
    enlarged_path = output_dir / "candidate_8x.png"

    with csv_path.open("w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file, lineterminator="\n")
        writer.writerows(matrix.tolist())

    pixels = (matrix * 255).astype(np.uint8)
    original = Image.fromarray(pixels, mode="L")
    original.save(original_path)

    try:
        nearest = Image.Resampling.NEAREST
    except AttributeError:
        nearest = Image.NEAREST

    enlarged = original.resize((1440, 720), resample=nearest)
    enlarged.save(enlarged_path)

    return csv_path, original_path, enlarged_path


def validate(matrix, csv_path, original_path, enlarged_path):
    assert matrix.shape == (90, 180)
    assert matrix.size == 16200
    assert set(np.unique(matrix).tolist()) <= {0, 1}

    with csv_path.open("r", newline="", encoding="utf-8") as file:
        rows = list(csv.reader(file))

    assert len(rows) == 90
    assert all(len(row) == 180 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(len(row) for row in rows) == 16200

    with Image.open(original_path) as image:
        assert image.size == (180, 90)
        assert set(image.convert("L").getdata()) <= {0, 255}

    with Image.open(enlarged_path) as image:
        assert image.size == (1440, 720)
        enlarged_array = np.asarray(image.convert("L"))

    expected = np.repeat(
        np.repeat(matrix * 255, 8, axis=0),
        8,
        axis=1
    ).astype(np.uint8)

    assert np.array_equal(enlarged_array, expected)


def main():
    matrix = build_matrix()
    paths = save_outputs(matrix)
    validate(matrix, *paths)

    print("Generated and validated:")
    for path in paths:
        print(path)
    print(
        f"Matrix: {matrix.shape[0]}x{matrix.shape[1]}, "
        f"values={matrix.size}, land={int(matrix.sum())}, "
        f"water={int(matrix.size - matrix.sum())}"
    )


if __name__ == "__main__":
    main()
