def inside_polygon(x, y, polygon):
    """Ray-crossing point-in-polygon test."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    # Coordinates are generated from the specified endpoints and 2/126-degree
    # spacing, with north-to-south rows and west-to-east columns.
    latitudes = np.linspace(55.0, 53.0, 127)
    longitudes = np.linspace(-131.0, -129.0, 127)
    matrix = np.zeros((127, 127), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            on_land = any(inside_polygon(lon, lat, poly) for poly in MAINLAND)
            on_island = any(inside_polygon(lon, lat, poly) for poly in ISLANDS)
            matrix[row, col] = int(on_land or on_island)

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Land is white (255); water is black (0).

    image.save(out / "candidate.png")
    image.resize((1016, 1016), resample=Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )


if __name__ == "__main__":
    main()
