def inside_polygon(vertices):
    """Vectorized point-in-polygon test for all sample coordinates."""
    inside = np.zeros(xx.shape, dtype=bool)
    n = len(vertices)
    for i in range(n):
        x1, y1 = vertices[i]
        x2, y2 = vertices[(i + 1) % n]
        crosses = (y1 > yy) != (y2 > yy)
        x_cross = (x2 - x1) * (yy - y1) / ((y2 - y1) or 1e-15) + x1
        inside ^= crosses & (xx < x_cross)
    return inside

for polygon in polygons:
    land |= inside_polygon(polygon)
