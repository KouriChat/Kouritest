我会按你指定的无外部地理数据方式，用手写海岸边界和岛屿范围规则生成采样矩阵；同时遵守本轮较新的直接答复要求，在最终消息中给出完整脚本，不尝试工具或写入文件。
下面脚本仅用手写的近似海岸线折线、岛屿轮廓及多边形点包含规则分类；不是对 16,129 个点逐点独立推理，也不调用外部地图或地理数据。海岸线与岛屿位置为基于已有地理知识的近似，细小岛礁可能遗漏。运行依赖为 Python、NumPy 和 Pillow；脚本会在当前目录创建 `outputs/` 并写出三项结果。

from pathlib import Path

LAND_POLYGONS = [

def point_in_polygon(lon, lat, polygon):
    """Ray-casting point test; boundary points are treated as land."""
    inside = False
    count = len(polygon)
    for i in range(count):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % count]

        # Explicitly include points on an edge.
        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-12:
            if min(x1, x2) - 1e-12 <= lon <= max(x1, x2) + 1e-12:
                if min(y1, y2) - 1e-12 <= lat <= max(y1, y2) + 1e-12:
                    return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < crossing_lon:
                inside = not inside
    return inside


def build_matrix():
    latitudes = np.linspace(LAT_NORTH, LAT_SOUTH, SIZE)
    longitudes = np.linspace(LON_WEST, LON_EAST, SIZE)
    matrix = np.zeros((SIZE, SIZE), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(
                any(point_in_polygon(lon, lat, polygon)
                    for polygon in LAND_POLYGONS)
            )
    return matrix


def main():
    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()
    if matrix.shape != (127, 127) or matrix.size != 16129:
        raise RuntimeError("Unexpected matrix dimensions.")
    if not np.isin(matrix, [0, 1]).all():
        raise RuntimeError("Matrix contains values other than 0 and 1.")

    np.savetxt(out_dir / "candidate.csv", matrix, fmt="%d", delimiter=",")

    # Matrix value 1 (land) is white; 0 (water) is black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out_dir / "candidate.png")

    enlarged = image.resize((1016, 1016), resample=Image.Resampling.NEAREST)
    enlarged.save(out_dir / "candidate_8x.png")

    # Verify CSV shape/content and image dimensions/palette.
    rows = (out_dir / "candidate.csv").read_text(encoding="ascii").splitlines()
    if len(rows) != 127 or any(
        len(line.split(",")) != 127
        or any(value not in ("0", "1") for value in line.split(","))
        for line in rows
    ):
        raise RuntimeError("CSV validation failed.")

    raw = np.asarray(Image.open(out_dir / "candidate.png"))
    scaled = np.asarray(Image.open(out_dir / "candidate_8x.png"))
    if raw.shape != (127, 127) or not set(np.unique(raw)).issubset({0, 255}):
        raise RuntimeError("Original PNG validation failed.")
    if scaled.shape != (1016, 1016):
        raise RuntimeError("Enlarged PNG dimensions are incorrect.")
    if not np.array_equal(scaled, np.repeat(np.repeat(raw, 8, axis=0), 8, axis=1)):
        raise RuntimeError("Enlarged PNG is not an exact nearest-neighbor scale.")

    print("Wrote validated 127x127 matrix and PNGs to outputs/.")


if __name__ == "__main__":
    main()
