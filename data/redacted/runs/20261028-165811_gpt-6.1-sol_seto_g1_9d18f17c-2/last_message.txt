from pathlib import Path

ROWS, COLS = 64, 253

STEP = 2.0 / 63.0

def inside_polygon(lon, lat, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            cross = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < cross:
                inside = not inside
        j = i
    return inside

def is_land(lon, lat):
    if any(inside_polygon(lon, lat, poly) for poly in LAND_POLYGONS):
        return True
    for cx, cy, rx, ry in ISLANDS:
        if ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0:
            return True
    return False

def main():
    lats = LAT_TOP - np.arange(ROWS) * STEP
    lons = LON_LEFT + np.arange(COLS) * STEP
    matrix = np.zeros((ROWS, COLS), dtype=np.uint8)

    for r, lat in enumerate(lats):
        for c, lon in enumerate(lons):
            matrix[r, c] = int(is_land(float(lon), float(lat)))

    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    pixels = np.where(matrix == 1, 255, 0).astype(np.uint8)
    image = Image.fromarray(pixels, mode="L")
    image.save(out / "candidate.png")
    image.resize((COLS * 8, ROWS * 8), Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )

if __name__ == "__main__":
    main()
