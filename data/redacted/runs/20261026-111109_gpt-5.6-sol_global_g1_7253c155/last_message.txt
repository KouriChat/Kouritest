def inside_polygon(lon, lat, polygon):
    """Ray-casting point-in-polygon test."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_x:
                inside = not inside
        j = i
    return inside

def antarctica_coast_lat(lon):
    # Approximate southern-continent coast; interpolate a hand-written outline.
    outline = [
        (-180, -69), (-160, -71), (-140, -70), (-120, -73),
        (-100, -72), (-80, -75), (-60, -69), (-50, -64),
        (-40, -68), (-20, -72), (0, -72), (20, -70),
        (40, -69), (60, -67), (80, -67), (100, -66),
        (120, -66), (140, -67), (160, -69), (180, -69),
    ]
    xs = np.array([p[0] for p in outline], dtype=float)
    ys = np.array([p[1] for p in outline], dtype=float)
    return float(np.interp(lon, xs, ys))

    land = np.zeros((len(lats), len(lons)), dtype=np.uint8)

    for row, lat in enumerate(lats):

            is_land = any(inside_polygon(lon, lat, poly) for poly in POLYGONS)
            if lat <= antarctica_coast_lat(lon):
                is_land = True
            land[row, col] = int(is_land)

    out = Path("outputs")
