def point_in_polygon(lon, lat, polygon):
    """Ray-crossing test for one point and a hand-written polygon."""
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            cross_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < cross_x:
                inside = not inside
        j = i
    return inside

def point_on_island(lon, lat, island):
    cx, cy, rx, ry = island
    if rx <= 0 or ry <= 0:
        return False
    return ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0

def is_land(lon, lat):
    if any(point_in_polygon(lon, lat, polygon) for polygon in POLYGONS):
        return True
    return any(point_on_island(lon, lat, island) for island in ISLANDS)

def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = np.zeros((len(LATITUDES), len(LONGITUDES)), dtype=np.uint8)
    for row, lat in enumerate(LATITUDES):
        for col, lon in enumerate(LONGITUDES):
            matrix[row, col] = int(is_land(float(lon), float(lat)))

    assert matrix.shape == (71, 231)
    assert matrix.size == 16401
    assert np.isin(matrix, (0, 1)).all()

    with (output_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Convert 0 -> black water and 1 -> white land.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    assert image.size == (231, 71)
    assert set(image.getdata()).issubset({(0, 0, 0), (255, 255, 255)})
    image.save(output_dir / "candidate.png")

    enlarged = image.resize((231 * 8, 71 * 8), resample=Image.Resampling.NEAREST)
    assert enlarged.size == (1848, 568)
    enlarged.save(output_dir / "candidate_8x.png")

    # Validate the emitted CSV and nearest-neighbor pixel replication.
    with (output_dir / "candidate.csv").open("r", newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))
    assert len(rows) == 71
    assert all(len(row) == 231 and all(value in {"0", "1"} for value in row) for row in rows)
    assert np.array_equal(np.asarray(enlarged)[::8, ::8], np.asarray(image))

if __name__ == "__main__":
    main()
