def points_in_polygon(lon_grid, lat_grid, polygon):
    """Vectorized ray-casting point-in-polygon test."""
    inside = np.zeros(lon_grid.shape, dtype=bool)
    x = lon_grid
    y = lat_grid
    vertices = polygon
    j = len(vertices) - 1
    for i in range(len(vertices)):
        xi, yi = vertices[i]
        xj, yj = vertices[j]
        crosses = (yi > y) != (yj > y)
        boundary_x = (xj - xi) * (y - yi) / ((yj - yi) + 1e-30) + xi
        inside ^= crosses & (x < boundary_x)
        j = i
    return inside


def build_matrix():
    latitudes = 23.0 - 0.2 * np.arange(111)
    longitudes = 99.0 + 0.2 * np.arange(141)
    lon_grid, lat_grid = np.meshgrid(longitudes, latitudes)
    land = np.zeros((111, 141), dtype=bool)
    for polygon in POLYGONS:
        land |= points_in_polygon(lon_grid, lat_grid, polygon)
    return land.astype(np.uint8)


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()
    if matrix.shape != (111, 141):
        raise RuntimeError(f"Unexpected matrix shape: {matrix.shape}")

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Image mode "1" is a binary black-and-white image; land is white.
    image = Image.fromarray((matrix * 255).astype(np.uint8), mode="L")
    if image.size != (141, 111):
        raise RuntimeError(f"Unexpected image size: {image.size}")
    image.save(output_dir / "candidate.png")

    enlarged = image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    # Verify the emitted CSV and image dimensions and values.
    rows = csv_path.read_text(encoding="ascii").splitlines()
    if len(rows) != 111 or any(
        len(row.split(",")) != 141
        or any(value not in {"0", "1"} for value in row.split(","))
        for row in rows
    ):
        raise RuntimeError("CSV validation failed")
    if set(np.unique(np.asarray(image))) != {0, 255}:
        raise RuntimeError("Image is not strictly black and white")
    if Image.open(output_dir / "candidate_8x.png").size != (1128, 888):
        raise RuntimeError("Enlarged image size validation failed")


if __name__ == "__main__":
    main()
