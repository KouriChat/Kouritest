def on_segment(x, y, a, b, eps=1e-12):
    cross = (x - a[0]) * (b[1] - a[1]) - (y - a[1]) * (b[0] - a[0])
    if abs(cross) > eps:
        return False
    return (
        min(a[0], b[0]) - eps <= x <= max(a[0], b[0]) + eps
        and min(a[1], b[1]) - eps <= y <= max(a[1], b[1]) + eps
    )


def inside_polygon(x, y, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if on_segment(x, y, (xi, yi), (xj, yj)):
            return True
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    if any(inside_polygon(lon, lat, poly) for poly in LAND_POLYGONS):
        return True
    for cx, cy, rx, ry in ISLANDS:
        if ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1:
            return True
    return False


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    rows = []
    for r in range(64):
        lat = 35.0 - r * (2.0 / 63.0)
        row = []
        for c in range(253):
            lon = 129.0 + c * (8.0 / 252.0)
            row.append(1 if is_land(lon, lat) else 0)
        rows.append(row)

    matrix = np.asarray(rows, dtype=np.uint8)
    with (out / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())
