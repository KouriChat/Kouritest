def point_on_segment(x, y, x1, y1, x2, y2):
    cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
    if abs(cross) > 1e-12:
        return False

    return (
        min(x1, x2) - 1e-12 <= x <= max(x1, x2) + 1e-12
        and min(y1, y2) - 1e-12 <= y <= max(y1, y2) + 1e-12
    )


def point_in_polygon(lon, lat, polygon):
    inside = False
    n = len(polygon)

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        if point_on_segment(lon, lat, x1, y1, x2, y2):
            return True

        crosses_latitude = (y1 > lat) != (y2 > lat)
        if crosses_latitude:
            intersection_lon = (
                (x2 - x1) * (lat - y1) / (y2 - y1) + x1
            )
            if lon < intersection_lon:
                inside = not inside

    return inside


def is_land(lon, lat):
    for polygon in LAND_POLYGONS:
        if point_in_polygon(lon, lat, polygon):
            for lake in LAKE_POLYGONS:
                if point_in_polygon(lon, lat, lake):
                    return 0
            return 1

    return 0


def build_matrix():
    matrix = np.zeros((ROWS, COLS), dtype=np.uint8)

    for row, lat in enumerate(LATITUDES):
        for col, lon in enumerate(LONGITUDES):
            matrix[row, col] = is_land(float(lon), float(lat))

    return matrix


def write_csv(matrix, path):
    with path.open("w", newline="", encoding="ascii") as handle:
        writer = csv.writer(handle, lineterminator="\n")
        for row in matrix:
            writer.writerow([str(int(value)) for value in row])


def write_images(matrix, original_path, enlarged_path):
    # 0 表示水域，转换为黑色；1 表示陆地，转换为白色。
    pixels = matrix * 255
    image = Image.fromarray(pixels.astype(np.uint8), mode="L")
    image.save(original_path)

    enlarged = image.resize((1440, 720), resample=Image.Resampling.NEAREST)
    enlarged.save(enlarged_path)


def validate(matrix, csv_path, original_path, enlarged_path):
    if matrix.shape != (90, 180):
        raise RuntimeError(f"矩阵尺寸错误：{matrix.shape}")

    values = set(int(value) for value in matrix.ravel())
    if not values.issubset({0, 1}):
        raise RuntimeError(f"矩阵包含非法值：{values}")

    if matrix.size != 16200:
        raise RuntimeError(f"有效值数量错误：{matrix.size}")

    with csv_path.open("r", encoding="ascii") as handle:
        lines = handle.read().splitlines()

    if len(lines) != 90:
        raise RuntimeError("CSV 行数错误")

    for line in lines:
        fields = line.split(",")
        if len(fields) != 180:
            raise RuntimeError("CSV 列数错误")
        if any(field not in {"0", "1"} for field in fields):
            raise RuntimeError("CSV 含有非 0/1 值")

    with Image.open(original_path) as image:
        if image.size != (180, 90):
            raise RuntimeError(f"原始图片尺寸错误：{image.size}")

        colors = set(image.getdata())
        if not colors.issubset({0, 255}):
            raise RuntimeError(f"原始图片含有非黑白像素：{colors}")

    with Image.open(enlarged_path) as image:
        if image.size != (1440, 720):
            raise RuntimeError(f"放大图片尺寸错误：{image.size}")

        colors = set(image.getdata())
        if not colors.issubset({0, 255}):
            raise RuntimeError(f"放大图片含有非黑白像素：{colors}")


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    original_path = output_dir / "candidate.png"
    enlarged_path = output_dir / "candidate_8x.png"

    matrix = build_matrix()
    write_csv(matrix, csv_path)
    write_images(matrix, original_path, enlarged_path)
    validate(matrix, csv_path, original_path, enlarged_path)

    print("生成完成：")
    print(f"  {csv_path}")
    print(f"  {original_path}")
    print(f"  {enlarged_path}")
    print("矩阵尺寸：90 x 180")
    print("有效值数量：16,200")
    print("取值范围：0/1")
    print("图片放大方式：最近邻，8 倍")


if __name__ == "__main__":
    main()
