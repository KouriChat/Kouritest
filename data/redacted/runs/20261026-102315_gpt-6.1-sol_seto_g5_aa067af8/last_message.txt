def polygon(points):
    return [(float(x), float(y)) for x, y in points]


def ellipse(cx, cy, rx, ry, count=20, phase=0.0):
    pts = []
    for i in range(count):
        a = phase + 2.0 * math.pi * i / count
        pts.append((cx + rx * math.cos(a), cy + ry * math.sin(a)))
    return pts


def inside(x, y, points):
    """Ray-crossing point-in-polygon test."""
    result = False
    j = len(points) - 1

    for i in range(len(points)):
        xi, yi = points[i]
        xj, yj = points[j]

        crosses = ((yi > y) != (yj > y))
        if crosses:
            at_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < at_x:
                result = not result

        j = i

    return result

def classify(lon, lat):
    for shape in land_polygons:
        if inside(lon, lat, shape):
            return 1
    return 0


def build_matrix():
    matrix = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)

    for row in range(HEIGHT):
        lat = LAT_MAX - row * STEP
        for col in range(WIDTH):
            lon = LON_MIN + col * STEP
            matrix[row, col] = classify(lon, lat)

    return matrix


def write_outputs(matrix):
    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    csv_path = out_dir / "candidate.csv"
    png_path = out_dir / "candidate.png"
    enlarged_path = out_dir / "candidate_8x.png"

    np.savetxt(csv_path, matrix, fmt="%d", delimiter=",")

    # Land is white and water is black.
    pixels = (matrix * 255).astype(np.uint8)
    original = Image.fromarray(pixels)
    original.save(png_path)

    enlarged = original.resize(
        (WIDTH * 8, HEIGHT * 8),
        resample=Image.Resampling.NEAREST,
    )
    enlarged.save(enlarged_path)

    return csv_path, png_path, enlarged_path


def validate(matrix, paths):
    assert matrix.shape == (64, 253)
    assert matrix.size == 16192
    assert set(np.unique(matrix).tolist()).issubset({0, 1})

    csv_path, png_path, enlarged_path = paths

    rows = csv_path.read_text(encoding="utf-8").splitlines()
    assert len(rows) == 64
    for row in rows:
        values = row.split(",")
        assert len(values) == 253
        assert all(value in {"0", "1"} for value in values)

    original = Image.open(png_path).convert("L")
    enlarged = Image.open(enlarged_path).convert("L")

    assert original.size == (253, 64)
    assert enlarged.size == (2024, 512)

    assert set(original.getdata()).issubset({0, 255})
    assert set(enlarged.getdata()).issubset({0, 255})

    original_array = np.asarray(original)
    enlarged_array = np.asarray(enlarged)

    expected = np.repeat(np.repeat(original_array, 8, axis=0), 8, axis=1)
    assert np.array_equal(enlarged_array, expected)


def main():
    matrix = build_matrix()
    paths = write_outputs(matrix)
    validate(matrix, paths)

    print("matrix:", matrix.shape, "valid values:", matrix.size)
    print("csv:", paths[0])
    print("png:", paths[1], Image.open(paths[1]).size)
    print("png_8x:", paths[2], Image.open(paths[2]).size)


if __name__ == "__main__":
    main()
