def point_in_polygon(lon, lat, polygon):
    """标准射线法判断点是否在多边形内部。"""
    inside = False
    n = len(polygon)

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        crosses = ((y1 > lat) != (y2 > lat))
        if crosses:
            x_at_lat = (x2 - x1) * (lat - y1) / (y2 - y1) + x1
            if lon < x_at_lat:
                inside = not inside

    return inside


def point_in_ellipse(lon, lat, ellipse):
    cx, cy, rx, ry = ellipse
    return ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0


def is_land(lon, lat):
    for polygon in POLYGONS:
        if point_in_polygon(lon, lat, polygon):
            return 1

    for ellipse in ELLIPSES:
        if point_in_ellipse(lon, lat, ellipse):
            return 1

    return 0


def build_matrix():
    matrix = []

    # 从北到南：89, 87, ..., -89
    # 从西到东：-179, -177, ..., 179
    for lat in range(89, -90, -2):
        row = []
        for lon in range(-179, 180, 2):
            row.append(is_land(lon, lat))
        matrix.append(row)

    return matrix


def write_csv(matrix):
    path = OUT / "candidate.csv"
    with path.open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix)


def write_images(matrix):
    height = len(matrix)
    width = len(matrix[0])

    # 黑色为水域，白色为陆地
    image = Image.new("1", (width, height), 0)
    pixels = image.load()

    for y, row in enumerate(matrix):
        for x, value in enumerate(row):
            pixels[x, y] = 1 if value == 1 else 0

    image.save(OUT / "candidate.png")

    enlarged = image.resize((width * 8, height * 8), Image.Resampling.NEAREST)
    enlarged.save(OUT / "candidate_8x.png")


def validate(matrix):
    assert len(matrix) == 90
    assert all(len(row) == 180 for row in matrix)
    assert sum(len(row) for row in matrix) == 16200
    assert all(value in (0, 1) for row in matrix for value in row)

    image = Image.open(OUT / "candidate.png").convert("1")
    enlarged = Image.open(OUT / "candidate_8x.png").convert("1")

    assert image.size == (180, 90)
    assert enlarged.size == (1440, 720)

    # 确认放大图确实为 8 倍最近邻复制
    for y in range(90):
        for x in range(180):
            expected = image.getpixel((x, y))
            for dy in range(8):
                for dx in range(8):
                    assert enlarged.getpixel((x * 8 + dx, y * 8 + dy)) == expected


def main():
    matrix = build_matrix()
    write_csv(matrix)
    write_images(matrix)
    validate(matrix)

    land_count = sum(value for row in matrix for value in row)
    water_count = 16200 - land_count

    print("生成完成")
    print(f"矩阵尺寸: {len(matrix)} x {len(matrix[0])}")
    print(f"有效值数量: {len(matrix) * len(matrix[0])}")
    print(f"陆地点数: {land_count}")
    print(f"水域点数: {water_count}")
    print("文件:")
    print("  outputs/candidate.csv")
    print("  outputs/candidate.png")
    print("  outputs/candidate_8x.png")


if __name__ == "__main__":
    main()

脚本会在当前目录创建：

- `outputs/candidate.csv`：90 行 × 180 列、无表头、仅含 `0` 和 `1`
- `outputs/candidate.png`：180×90 原始黑白图
- `outputs/candidate_8x.png`：1440×720、最近邻放大 8 倍图

其中陆地由手写多边形和椭圆规则的并集决定，未进行逐点独立推理、海岸线平滑或后处理修补。
