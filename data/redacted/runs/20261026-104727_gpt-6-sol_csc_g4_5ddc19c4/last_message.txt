def point_in_polygon(lon, lat, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > lat) != (yj > lat):
            crossing_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < crossing_lon:
                inside = not inside
        j = i
    return inside


def is_land(lon, lat):
    if any(point_in_polygon(lon, lat, poly) for poly in LAND_POLYGONS):
        return True
    for cx, cy, rx, ry in ISLANDS:
        if ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0:
            return True
    return False


def main():
    latitudes = [23.0 - 0.2 * i for i in range(111)]
    longitudes = [99.0 + 0.2 * i for i in range(141)]
    matrix = np.fromfunction(
        np.vectorize(lambda r, c: int(is_land(longitudes[int(c)], latitudes[int(r)]))),
        (111, 141),
        dtype=int,
    ).astype(np.uint8)

    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Matrix 1 (land) maps to white; 0 (water) maps to black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out_dir / "candidate.png")
    image.resize((141 * 8, 111 * 8), Image.Resampling.NEAREST).save(
        out_dir / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)

    assert Image.open(out_dir / "candidate.png").size == (141, 111)
    assert Image.open(out_dir / "candidate_8x.png").size == (1128, 888)


if __name__ == "__main__":
    main()
