def point_in_polygon(lon, lat, polygon):
    """Ray casting; boundary points are included as land."""
    inside = False
    n = len(polygon)
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-10:
            if (min(x1, x2) - 1e-10 <= lon <= max(x1, x2) + 1e-10
                    and min(y1, y2) - 1e-10 <= lat <= max(y1, y2) + 1e-10):
                return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < crossing_lon:
                inside = not inside
    return inside


def is_small_island_or_reef(lon, lat):
    for center_lon, center_lat, radius_km in ISLANDS_KM:
        # Longitude degrees shrink with latitude; this ellipse is a local
        # approximation to the specified radius.
        dy = (lat - center_lat) * 111.0
        dx = (lon - center_lon) * 111.0 * math.cos(math.radians(center_lat))
        if dx * dx + dy * dy <= radius_km * radius_km:
            return True
    return False


def is_land(lon, lat):
    return any(point_in_polygon(lon, lat, p) for p in POLYGONS) or \
        is_small_island_or_reef(lon, lat)


def main():
    latitudes = [23.0 - 0.2 * i for i in range(111)]
    longitudes = [99.0 + 0.2 * j for j in range(141)]
    matrix = np.zeros((111, 141), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(is_land(lon, lat))

    out_dir = Path("outputs")
    out_dir.mkdir(parents=True, exist_ok=True)

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Land is white; water is black.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(out_dir / "candidate.png")
    enlarged = image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(out_dir / "candidate_8x.png")

    assert matrix.shape == (111, 141)
    assert matrix.size == 15651
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)
    assert enlarged.size == (1128, 888)


if __name__ == "__main__":
    main()
