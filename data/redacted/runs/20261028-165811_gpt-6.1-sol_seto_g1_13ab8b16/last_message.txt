def point_on_segment(x, y, a, b, eps=1e-10):
    ax, ay = a
    bx, by = b
    cross = (x - ax) * (by - ay) - (y - ay) * (bx - ax)
    if abs(cross) > eps:
        return False
    return (min(ax, bx) - eps <= x <= max(ax, bx) + eps and
            min(ay, by) - eps <= y <= max(ay, by) + eps)

def inside_polygon(x, y, polygon):
    inside = False
    n = len(polygon)
    for i in range(n):
        a, b = polygon[i], polygon[(i + 1) % n]
        if point_on_segment(x, y, a, b):
            return True
        x1, y1 = a
        x2, y2 = b
        if (y1 > y) != (y2 > y):
            crossing_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing_x:
                inside = not inside
    return inside

def is_land(lon, lat):
    return any(inside_polygon(lon, lat, poly) for poly in POLYGONS)

matrix = np.zeros((ROWS, COLS), dtype=np.uint8)
for r in range(ROWS):
    lat = LAT_TOP - r * STEP
    for c in range(COLS):
        lon = LON_LEFT + c * STEP
        matrix[r, c] = int(is_land(lon, lat))
