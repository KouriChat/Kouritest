def point_on_segment(x, y, a, b, eps=1e-12):
    ax, ay = a
    bx, by = b
    cross = (x - ax) * (by - ay) - (y - ay) * (bx - ax)
    if abs(cross) > eps:
        return False
    return (min(ax, bx) - eps <= x <= max(ax, bx) + eps and
            min(ay, by) - eps <= y <= max(ay, by) + eps)


def inside_polygon(x, y, polygon):
    """Ray-casting point test; boundary points count as land."""
    inside = False
    previous = polygon[-1]
    for current in polygon:
        if point_on_segment(x, y, previous, current):
            return True
        x1, y1 = previous
        x2, y2 = current
        if (y1 > y) != (y2 > y):
            crossing_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing_x:
                inside = not inside
        previous = current
    return inside


def is_land(lon, lat):
    if inside_polygon(lon, lat, MAINLAND):
        return True
    if inside_polygon(lon, lat, GRAHAM):
        return True
    if inside_polygon(lon, lat, MORESBY):
        return True
    return any(inside_polygon(lon, lat, poly) for poly in ISLANDS + REEFS)


def build_matrix():
    matrix = np.zeros((N, N), dtype=np.uint8)
    for row in range(N):
        lat = LAT_TOP - row * STEP
        for col in range(N):
            lon = LON_LEFT + col * STEP
            matrix[row, col] = 1 if is_land(lon, lat) else 0
    return matrix
