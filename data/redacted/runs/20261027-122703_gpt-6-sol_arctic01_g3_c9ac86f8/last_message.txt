def inside_polygon(lon, lat, vertices):
    """向量化射线法；lon、lat 为同形状的采样点数组。"""
    inside = np.zeros(lon.shape, dtype=bool)
    previous_x, previous_y = vertices[-1]

    for current_x, current_y in vertices:
        if (previous_y > current_y):
            upper_x, upper_y = previous_x, previous_y
            lower_x, lower_y = current_x, current_y
        else:
            upper_x, upper_y = current_x, current_y
            lower_x, lower_y = previous_x, previous_y

        crosses_latitude = (lat >= lower_y) & (lat < upper_y)
        if upper_y != lower_y:
            crossing_x = lower_x + (
                (lat - lower_y) * (upper_x - lower_x)
                / (upper_y - lower_y)
            )
            inside ^= crosses_latitude & (lon < crossing_x)

        previous_x, previous_y = current_x, current_y

    return inside


def main():
    # 用整数步数构造坐标，避免累加 0.1 时的浮点漂移。
    longitudes = -105.0 + np.arange(241) / 10.0
    latitudes = 79.0 - np.arange(61) / 10.0
    lon, lat = np.meshgrid(longitudes, latitudes)

    land = np.zeros((61, 241), dtype=bool)
    for polygon in LAND_POLYGONS:
        land |= inside_polygon(lon, lat, polygon)
    for polygon in WATER_POLYGONS:
        land &= ~inside_polygon(lon, lat, polygon)

    matrix = land.astype(np.uint8)
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    original_path = output_dir / "candidate.png"
    enlarged_path = output_dir / "candidate_8x.png"

    with csv_path.open("w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        writer.writerows(matrix.tolist())

    original = Image.fromarray(matrix * 255, mode="L")
    original.save(original_path)
    nearest = getattr(getattr(Image, "Resampling", Image), "NEAREST")
    original.resize((1928, 488), resample=nearest).save(enlarged_path)

    # 从交付文件重新读取并核验。
    with csv_path.open(newline="", encoding="utf-8") as file:
        rows = list(csv.reader(file))
    assert len(rows) == 61
    assert all(len(row) == 241 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(map(len, rows)) == 14_701
    assert np.array_equal(
        np.asarray(rows, dtype=np.uint8), matrix
    )

    with Image.open(original_path) as image:
        raw = np.asarray(image.convert("L"))
        assert image.size == (241, 61)
        assert set(np.unique(raw).tolist()) <= {0, 255}
        assert np.array_equal(raw, matrix * 255)

    with Image.open(enlarged_path) as image:
        enlarged = np.asarray(image.convert("L"))
        assert image.size == (1928, 488)
        assert set(np.unique(enlarged).tolist()) <= {0, 255}
        assert np.array_equal(
            enlarged, np.repeat(np.repeat(raw, 8, axis=0), 8, axis=1)
        )

    print("已生成并验证：")
    print(csv_path)
    print(original_path)

if __name__ == "__main__":
