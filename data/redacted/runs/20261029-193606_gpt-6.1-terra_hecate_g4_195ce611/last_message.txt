def contains(polygon, x, y):
    """Ray casting; boundary points are assigned according to this rule."""
    inside = False
    previous_x, previous_y = polygon[-1]
    for current_x, current_y in polygon:
        if (current_y > y) != (previous_y > y):
            crossing_x = (
                (previous_x - current_x)
                * (y - current_y)
                / (previous_y - current_y)
                + current_x
            )
            if x < crossing_x:
                inside = not inside
        previous_x, previous_y = current_x, current_y
    return inside


def main():
    size = 127
    scale = 8
    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)

    matrix = []
    for row in range(size):

        values = []
        for col in range(size):
            longitude = -131.0 + 2.0 * col / (size - 1)
            land = any(contains(poly, longitude, latitude) for poly in LAND)
            water = any(contains(poly, longitude, latitude) for poly in WATER)
            values.append(int(land and not water))
        matrix.append(values)

    csv_path = output / "candidate.csv"

        csv.writer(stream, lineterminator="\n").writerows(matrix)

    image = Image.new("L", (size, size))
    image.putdata([255 * value for row in matrix for value in row])
    image.save(output / "candidate.png")

    enlarged = image.resize(
        (size * scale, size * scale), resample=Image.Resampling.NEAREST
    )
    enlarged.save(output / "candidate_8x.png")

    # Re-read the deliverables and verify shape, values, colors, and scaling.
    with csv_path.open(newline="", encoding="ascii") as stream:
        rows = list(csv.reader(stream))
    assert len(rows) == size
    assert all(len(row) == size for row in rows)
    assert all(value in ("0", "1") for row in rows for value in row)
    assert sum(len(row) for row in rows) == 16129

    with Image.open(output / "candidate.png") as original:
        assert original.size == (size, size)
        assert original.mode == "L"
        assert set(original.getdata()) <= {0, 255}
        assert list(original.getdata()) == [
            int(value) * 255 for row in rows for value in row
        ]

        with Image.open(output / "candidate_8x.png") as large:
            assert large.size == (1016, 1016)

            assert set(large.getdata()) <= {0, 255}
            for y in range(large.height):
                for x in range(large.width):
                    assert large.getpixel((x, y)) == original.getpixel(
                        (x // scale, y // scale)
                    )

    print("Verified 127 x 127 matrix and both PNG files in outputs/.")
