def point_in_polygon(lon, lat, polygon):
    """Ray casting point test; boundary points are treated as land."""
    inside = False
    count = len(polygon)
    for i in range(count):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % count]

        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if (
            abs(cross) < 1e-10
            and min(x1, x2) - 1e-10 <= lon <= max(x1, x2) + 1e-10
            and min(y1, y2) - 1e-10 <= lat <= max(y1, y2) + 1e-10
        ):
            return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < crossing_lon:
                inside = not inside
    return inside


def main():
    latitudes = np.round(np.arange(79.0, 72.9, -0.1), 1)
    longitudes = np.round(np.arange(-105.0, -80.9, 0.1), 1)

    matrix = np.zeros((len(latitudes), len(longitudes)), dtype=np.uint8)
    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            matrix[row, col] = int(
                any(point_in_polygon(lon, lat, polygon)
                    for polygon in LAND_POLYGONS)
            )

    if matrix.shape != (61, 241):
        raise RuntimeError(f"Unexpected matrix shape: {matrix.shape}")

    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    with (output_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # RGB values are exactly black for water and white for land.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    image.resize((241 * 8, 61 * 8), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    values = np.asarray(image)
    if values.shape != (61, 241, 3) or not np.isin(matrix, (0, 1)).all():
        raise RuntimeError("Matrix or original image validation failed")
    if set(map(tuple, np.unique(values.reshape(-1, 3), axis=0))) != {
        (0, 0, 0), (255, 255, 255)
    }:
        raise RuntimeError("Original image contains unexpected colors")

    enlarged = Image.open(output_dir / "candidate_8x.png")
    if enlarged.size != (1928, 488):
        raise RuntimeError(f"Unexpected enlarged image size: {enlarged.size}")

    with (output_dir / "candidate.csv").open("r", encoding="ascii") as f:
        rows = [line.rstrip("\n").split(",") for line in f]
    if len(rows) != 61 or any(
        len(row) != 241 or any(value not in ("0", "1") for value in row)
        for row in rows
    ):
        raise RuntimeError("CSV validation failed")

    print("Validated 61x241 matrix, CSV, 241x61 PNG, and 1928x488 nearest-neighbor PNG.")


if __name__ == "__main__":
    main()
