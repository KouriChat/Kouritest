def contained(west, north, vertices):
    """向量化射线法；输入和输出均对应采样点，而非格子面积。"""
    inside = np.zeros(west.shape, dtype=bool)
    previous_w, previous_n = vertices[-1]

    for current_w, current_n in vertices:
        if previous_n != current_n:
            crosses_latitude = (
                (previous_n > north) != (current_n > north)
            )
            crossing_w = previous_w + (
                (current_w - previous_w)
                * (north - previous_n)
                / (current_n - previous_n)
            )
            inside ^= crosses_latitude & (west < crossing_w)

        previous_w, previous_n = current_w, current_n

    return inside


def main():
    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)

    # 使用整数十分度确定坐标，避免反复累加浮点步长。
    longitudes_west = np.arange(1050, 809, -1, dtype=np.int32) / 10.0
    latitudes_north = np.arange(790, 729, -1, dtype=np.int32) / 10.0
    west, north = np.meshgrid(longitudes_west, latitudes_north)

    land = np.zeros((61, 241), dtype=bool)
    for polygon in polygons:
        land |= contained(west, north, polygon)

    matrix = land.astype(np.uint8)
    csv_path = output / "candidate.csv"
    with csv_path.open("w", newline="", encoding="utf-8") as file:
        csv.writer(file).writerows(matrix.tolist())

    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output / "candidate.png")

    nearest = getattr(Image, "Resampling", Image).NEAREST
    enlarged = image.resize((1928, 488), resample=nearest)
    enlarged.save(output / "candidate_8x.png")

    # 检查磁盘上的交付物，而不只检查内存变量。
    with csv_path.open(newline="", encoding="utf-8") as file:
        rows = list(csv.reader(file))
    assert len(rows) == 61
    assert all(len(row) == 241 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(map(len, rows)) == 14701
    assert np.array_equal(
        np.asarray(rows, dtype=np.uint8), matrix
    )

    with Image.open(output / "candidate.png") as saved:
        original_pixels = np.asarray(saved.convert("RGB"))
        assert saved.size == (241, 61)

    with Image.open(output / "candidate_8x.png") as saved:
        enlarged_pixels = np.asarray(saved.convert("RGB"))
        assert saved.size == (1928, 488)

    colors = np.unique(original_pixels.reshape(-1, 3), axis=0)
    assert all(
        tuple(color) in {(0, 0, 0), (255, 255, 255)}
        for color in colors
    )
    assert np.array_equal(
        enlarged_pixels,
        np.repeat(np.repeat(original_pixels, 8, axis=0), 8, axis=1),
    )
    print("已生成并验证 outputs/ 中的 CSV、原始 PNG 和 8 倍 PNG。")


if __name__ == "__main__":
    main()
