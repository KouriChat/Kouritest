def point_in_polygon(x, y, polygon):
    """Ray casting; boundary points are treated as land."""
    inside = False
    n = len(polygon)
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if (
            abs(cross) < 1e-10
            and min(x1, x2) - 1e-10 <= x <= max(x1, x2) + 1e-10
            and min(y1, y2) - 1e-10 <= y <= max(y1, y2) + 1e-10
        ):
            return True

        if (y1 > y) != (y2 > y):
            crossing_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing_x:
                inside = not inside
    return inside


def main():
    latitudes = 23.0 - 0.2 * np.arange(111)
    longitudes = 99.0 + 0.2 * np.arange(141)
    matrix = np.zeros((111, 141), dtype=np.uint8)

    for row, latitude in enumerate(latitudes):
        for col, longitude in enumerate(longitudes):
            matrix[row, col] = int(
                any(point_in_polygon(float(longitude), float(latitude), p)
                    for p in POLYGONS)
            )

    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", encoding="ascii", newline="") as stream:
        for row in matrix:
            stream.write(",".join(str(int(value)) for value in row) + "\n")

    # Matrix value 1 is white land; 0 is black water.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    image.resize((1128, 888), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)
    assert int(matrix.size) == 15651
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)
    assert set(image.getdata()).issubset({(0, 0, 0), (255, 255, 255)})
    assert Image.open(output_dir / "candidate_8x.png").size == (1128, 888)


if __name__ == "__main__":
