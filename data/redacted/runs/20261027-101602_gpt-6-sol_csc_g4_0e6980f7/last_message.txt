def inside(lon, lat, vertices):
    """Even-odd point-in-polygon test."""
    result = False
    previous = vertices[-1]
    for current in vertices:
        x1, y1 = previous
        x2, y2 = current
        if (y1 > lat) != (y2 > lat):
            crossing = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon <= crossing:
                result = not result
        previous = current
    return result


def ellipse(lon, lat, center_x, center_y, radius_x, radius_y):
    return (
        ((lon - center_x) / radius_x) ** 2
        + ((lat - center_y) / radius_y) ** 2
        <= 1
    )

def is_land(lon, lat):
    return any(inside(lon, lat, shape) for shape in POLYGONS) or any(
        ellipse(lon, lat, *island) for island in ISLANDS
    )


def main():
    OUT.mkdir(exist_ok=True)

    rows = [
        [
            int(is_land(99 + column / 5, 23 - row / 5))
            for column in range(WIDTH)
        ]
        for row in range(HEIGHT)
    ]

    csv_path = OUT / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as stream:
        csv.writer(stream).writerows(rows)

    image = Image.new("L", (WIDTH, HEIGHT))
    image.putdata([255 * value for row in rows for value in row])
    image.save(OUT / "candidate.png")

    nearest = getattr(Image, "Resampling", Image).NEAREST
    enlarged = image.resize((WIDTH * 8, HEIGHT * 8), resample=nearest)
    enlarged.save(OUT / "candidate_8x.png")

    assert len(rows) == 111
    assert all(len(row) == 141 and set(row) <= {0, 1} for row in rows)
    with csv_path.open(newline="", encoding="ascii") as stream:
        saved = list(csv.reader(stream))
    assert len(saved) == 111
    assert all(len(row) == 141 and set(row) <= {"0", "1"} for row in saved)
    assert image.size == (141, 111)
    assert enlarged.size == (1128, 888)
    assert set(image.getdata()) <= {0, 255}
    assert set(enlarged.getdata()) <= {0, 255}
    assert all(
        enlarged.getpixel((x, y)) == image.getpixel((x // 8, y // 8))
        for y in range(888)
        for x in range(1128)
    )


if __name__ == "__main__":
    main()
