def point_in_polygon(x, y, polygon):
    inside = False
    j = len(polygon) - 1
    for i, (xi, yi) in enumerate(polygon):
        xj, yj = polygon[j]
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def classify(lon, lat):
    land = any(point_in_polygon(lon, lat, poly) for poly in LAND_POLYGONS)
    if any(point_in_polygon(lon, lat, poly) for poly in WATER_POLYGONS):
        land = False
    if any(
        ((lon - x) / rx) ** 2 + ((lat - y) / ry) ** 2 <= 1.0
        for x, y, rx, ry in ISLANDS
    ):
        land = True
    return int(land)

        values = []
        for col in range(253):
            lon = 129.0 + col * (8.0 / 252.0)
            values.append(classify(lon, lat))
        rows.append(values)

    with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(rows)

    assert set(image.getdata()) <= {0, 255}
    assert set(enlarged.getdata()) <= {0, 255}


if __name__ == "__main__":
