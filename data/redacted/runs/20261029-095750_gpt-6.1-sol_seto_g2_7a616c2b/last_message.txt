方法：脚本以手写的近似海岸线多边形和岛屿椭圆作为地理规则，对每个采样点逐点做点内判定；结果是这些规则的计算输出，并非对 16,192 个点分别进行独立地理推理。未使用外部地图或地理数据。运行依赖：Python、NumPy、Pillow。

#!/usr/bin/env python3
"""按手写近似轮廓生成濑户内海及周边海陆栅格。

不读取、下载或查询任何地理数据。海岸和岛屿仅为手写近似，
每个采样点依据多边形/椭圆规则判定，不做面积平均或多数表决。
"""
from pathlib import Path

# 固定采样范围。使用精确的 2/63 度间隔，端点包含在内。
NROWS, NCOLS = 64, 253
STEP = 2.0 / 63.0
LATS = 35.0 - np.arange(NROWS) * STEP
LONS = 129.0 + np.arange(NCOLS) * STEP


def polygon_contains(lon, lat, vertices):
    """射线法：判断点是否在经纬度平面多边形内。"""
    inside = False
    x, y = lon, lat
    n = len(vertices)
    j = n - 1
    for i in range(n):
        xi, yi = vertices[i]
        xj, yj = vertices[j]
        # 半开区间规则避免顶点重复计数。
        crosses = (yi > y) != (yj > y)
        if crosses:
            x_cross = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < x_cross:
                inside = not inside
        j = i
    return inside


def ellipse_contains(lon, lat, cx, cy, rx, ry, angle_deg=0.0):
    """旋转椭圆近似小岛；半轴单位为经度、纬度度数。"""
    import math

    a = math.radians(angle_deg)
    dx, dy = lon - cx, lat - cy
    xr = dx * math.cos(a) + dy * math.sin(a)
    yr = -dx * math.sin(a) + dy * math.cos(a)
    return (xr / rx) ** 2 + (yr / ry) ** 2 <= 1.0

# 用规则构造点判定矩阵。边界在轮廓内侧按陆地处理。
matrix = np.zeros((NROWS, NCOLS), dtype=np.uint8)
for r, lat in enumerate(LATS):
    for c, lon in enumerate(LONS):
        land = any(polygon_contains(float(lon), float(lat), p)

        matrix[r, c] = 1 if land else 0
