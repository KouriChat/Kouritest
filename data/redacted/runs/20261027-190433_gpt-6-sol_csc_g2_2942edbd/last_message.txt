def polygon_mask(vertices):
    """Return grid points inside a polygon using an even-odd ray test."""
    inside = np.zeros(LON.shape, dtype=bool)
    x = LON
    y = LAT
    points = vertices
    j = len(points) - 1
    for i in range(len(points)):
        xi, yi = points[i]
        xj, yj = points[j]
        crosses = ((yi > y) != (yj > y)) & (
            x < (xj - xi) * (y - yi) / ((yj - yi) or 1e-30) + xi
        )
        inside ^= crosses
        j = i
    return inside


def build_matrix():

    for polygon in LAND_POLYGONS:
        matrix[polygon_mask(polygon)] = 1

    return matrix


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()
    if matrix.shape != (111, 141):
        raise RuntimeError(f"Unexpected matrix shape: {matrix.shape}")

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(value)) for value in row) + "\n")

    # Palette: 0 -> black water, 1 -> white land.
    image = Image.fromarray(matrix * 255, mode="L").convert("RGB")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")

    # Validate dimensions, contents, and nearest-neighbor enlargement.
    rows = csv_path.read_text(encoding="ascii").splitlines()
    if len(rows) != 111 or any(len(row.split(",")) != 141 for row in rows):
        raise RuntimeError("CSV dimensions are invalid")
    if any(value not in {"0", "1"} for row in rows for value in row.split(",")):
        raise RuntimeError("CSV contains a value other than 0 or 1")
    if set(image.getdata()) != {(0, 0, 0), (255, 255, 255)}:
        raise RuntimeError("Original PNG is not strictly black and white")
    if enlarged.size != (1128, 888):
        raise RuntimeError("Enlarged PNG dimensions are invalid")
    if enlarged.getpixel((8, 8)) != image.getpixel((1, 1)):
        raise RuntimeError("Nearest-neighbor enlargement check failed")

    print(f"Validated {matrix.size} values; outputs written to {output_dir}/")


if __name__ == "__main__":
    main()
