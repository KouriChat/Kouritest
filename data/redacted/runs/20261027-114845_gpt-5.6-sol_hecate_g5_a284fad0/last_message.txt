def ellipse_polygon(cx, cy, rx, ry, angle_degrees, count=24):
    angle = math.radians(angle_degrees)
    ca, sa = math.cos(angle), math.sin(angle)
    result = []
    for k in range(count):
        t = 2.0 * math.pi * k / count
        ex, ey = rx * math.cos(t), ry * math.sin(t)
        result.append((cx + ex * ca - ey * sa,
                       cy + ex * sa + ey * ca))
    return result


def points_in_polygon(x, y, polygon):
    """Vectorized even-odd point-in-polygon test."""
    inside = np.zeros(x.shape, dtype=bool)
    x1, y1 = polygon[-1]

    for x2, y2 in polygon:
        crosses = ((y1 > y) != (y2 > y))
        denominator = y2 - y1
        if denominator == 0:
            denominator = 1e-300
        crossing_x = (x2 - x1) * (y - y1) / denominator + x1
        inside ^= crosses & (x < crossing_x)
        x1, y1 = x2, y2

    return inside


def generate_matrix():
    longitudes = np.linspace(-131.0, -129.0, 127)
    latitudes = np.linspace(55.0, 53.0, 127)
    x, y = np.meshgrid(longitudes, latitudes)

    land = np.zeros((127, 127), dtype=bool)

    all_land = list(LAND_POLYGONS)
    all_land.extend(ellipse_polygon(*item) for item in ELLIPSES)

    for polygon in all_land:
        land |= points_in_polygon(x, y, polygon)

    for polygon in WATER_POLYGONS:
        land &= ~points_in_polygon(x, y, polygon)

    return land.astype(np.uint8)


def validate_csv(path):
    with path.open("r", newline="", encoding="ascii") as handle:
        rows = list(csv.reader(handle))

    assert len(rows) == 127
    assert all(len(row) == 127 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)
    assert sum(len(row) for row in rows) == 16129


def main():
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = generate_matrix()
    assert matrix.shape == (127, 127)
    assert matrix.size == 16129
    assert set(np.unique(matrix)).issubset({0, 1})

    csv_path = output_dir / "candidate.csv"
    png_path = output_dir / "candidate.png"
    large_path = output_dir / "candidate_8x.png"

    np.savetxt(csv_path, matrix, fmt="%d", delimiter=",")

    pixels = matrix * np.uint8(255)
    original = Image.fromarray(pixels, mode="L")
    original.save(png_path)

    try:
        nearest = Image.Resampling.NEAREST
    except AttributeError:
        nearest = Image.NEAREST

    enlarged = original.resize((1016, 1016), resample=nearest)
    enlarged.save(large_path)

    # Required validation.
    validate_csv(csv_path)

    with Image.open(png_path) as check:
        original_array = np.asarray(check)
        assert check.size == (127, 127)
        assert set(np.unique(original_array)).issubset({0, 255})

    with Image.open(large_path) as check:
        enlarged_array = np.asarray(check)
        assert check.size == (1016, 1016)
        assert set(np.unique(enlarged_array)).issubset({0, 255})

    expected_enlargement = np.repeat(
        np.repeat(pixels, 8, axis=0), 8, axis=1
    )
    assert np.array_equal(enlarged_array, expected_enlargement)

    print(f"Wrote {csv_path}")
    print(f"Wrote {png_path}")
    print(f"Wrote {large_path}")
    print(f"Land samples: {int(matrix.sum())}")
    print(f"Water samples: {int(matrix.size - matrix.sum())}")


if __name__ == "__main__":
    main()
