def point_in_polygon(lon, lat, polygon):
    """Ray-crossing point test; boundary points are treated as land."""
    inside = False
    count = len(polygon)
    for i in range(count):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % count]

        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-10 and min(x1, x2) <= lon <= max(x1, x2) \
                and min(y1, y2) <= lat <= max(y1, y2):
            return True

        if (y1 > lat) != (y2 > lat):
            crossing_lon = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < crossing_lon:
                inside = not inside
    return inside


def build_matrix():
    latitudes = [23.0 - 0.2 * i for i in range(111)]
    longitudes = [99.0 + 0.2 * j for j in range(141)]
    result = np.zeros((111, 141), dtype=np.uint8)

    for row, lat in enumerate(latitudes):
        for col, lon in enumerate(longitudes):
            result[row, col] = int(
                any(point_in_polygon(lon, lat, poly) for poly in LAND_POLYGONS)
            )
    return result


def main():
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    matrix = build_matrix()
    with (output_dir / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        f.writelines(",".join(map(str, row)) + "\n" for row in matrix)

    image = Image.fromarray(np.where(matrix == 1, 255, 0).astype(np.uint8), mode="L")
    image.save(output_dir / "candidate.png")
    image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST).save(
        output_dir / "candidate_8x.png"
    )

    assert matrix.shape == (111, 141)
    assert set(np.unique(matrix)).issubset({0, 1})
    assert image.size == (141, 111)


if __name__ == "__main__":
    main()
