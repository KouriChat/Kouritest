def contains(lon, lat, polygon):
    """Ray-casting point-in-polygon test, including dateline-safe longitude."""
    points = []
    for x, y in polygon:
        while x - lon > 180:
            x -= 360
        while x - lon < -180:
            x += 360
        points.append((x, y))

    inside = False
    x1, y1 = points[-1]
    for x2, y2 in points:
        if (y1 > lat) != (y2 > lat):
            crossing = (x2 - x1) * (lat - y1) / (y2 - y1) + x1
            if lon < crossing:
                inside = not inside
        x1, y1 = x2, y2
    return inside


def main():
    lats = np.arange(89, -90, -2)
    lons = np.arange(-179, 180, 2)
    matrix = np.zeros((len(lats), len(lons)), dtype=np.uint8)

    for row, lat in enumerate(lats):
        for col, lon in enumerate(lons):
            if any(contains(lon, lat, polygon) for polygon in POLYGONS):
                matrix[row, col] = 1
            if any(contains(lon, lat, lake) for lake in LAKES):
                matrix[row, col] = 0

    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    np.savetxt(out / "candidate.csv", matrix, fmt="%d", delimiter=",")

    # Matrix row zero is the northernmost latitude; white is land.
    pixels = np.where(matrix[:, :, None] == 1, 255, 0).astype(np.uint8)
    rgb = np.repeat(pixels, 3, axis=2)
    image = Image.fromarray(rgb, mode="RGB")
    image.save(out / "candidate.png")
    image.resize((1440, 720), resample=Image.Resampling.NEAREST).save(
        out / "candidate_8x.png"
    )


if __name__ == "__main__":
    main()
