def polygon_mask(vertices, lon_grid, lat_grid):
    """Even-odd point-in-polygon test on the sampling grid."""
    inside = np.zeros(lon_grid.shape, dtype=bool)
    x1, y1 = vertices[-1]
    for x2, y2 in vertices:
        if y1 != y2:
            crosses = (y1 > lat_grid) != (y2 > lat_grid)
            boundary_x = x1 + (lat_grid - y1) * (x2 - x1) / (y2 - y1)
            inside ^= crosses & (lon_grid < boundary_x)
        x1, y1 = x2, y2
    return inside


def main():
    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)

    latitudes = np.arange(89, -90, -2)
    longitudes = np.arange(-179, 180, 2)
    lon_grid, lat_grid = np.meshgrid(longitudes, latitudes)

    land = np.zeros((90, 180), dtype=bool)
    for polygon in POLYGONS:
        land |= polygon_mask(polygon, lon_grid, lat_grid)

    matrix = land.astype(np.uint8)
    assert matrix.shape == (90, 180)
    assert matrix.size == 16200
    assert np.isin(matrix, (0, 1)).all()

    csv_path = output / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    pixels = matrix * 255
    image = Image.fromarray(pixels, mode="L")
    image.save(output / "candidate.png")

    # Repeat pixels explicitly: each original pixel becomes an 8x8 block.
    enlarged_pixels = np.repeat(np.repeat(pixels, 8, axis=0), 8, axis=1)
    enlarged = Image.fromarray(enlarged_pixels, mode="L")
    enlarged.save(output / "candidate_8x.png")

    with csv_path.open(newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))
    assert len(rows) == 90
    assert all(len(row) == 180 for row in rows)
    assert all(value in ("0", "1") for row in rows for value in row)

    with Image.open(output / "candidate.png") as small_file:
        small = np.asarray(small_file)
        assert small_file.size == (180, 90)
        assert set(np.unique(small)).issubset({0, 255})
        assert np.array_equal(small, pixels)

    with Image.open(output / "candidate_8x.png") as large_file:
        large = np.asarray(large_file)
        assert large_file.size == (1440, 720)
        assert set(np.unique(large)).issubset({0, 255})
        assert np.array_equal(
            large, np.repeat(np.repeat(small, 8, axis=0), 8, axis=1)
        )

    print("Validated: 90 x 180 = 16,200 binary values.")
    print("Validated: 180 x 90 original and 1440 x 720 nearest-neighbor image.")
    print(f"Land points: {int(matrix.sum())}; water points: {int(matrix.size - matrix.sum())}")
    print("Files written to:", output.resolve())


if __name__ == "__main__":
    main()
