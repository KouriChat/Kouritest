# Water polygons remove broad approximations that would otherwise cover
# portions of the Black Sea and the Adriatic.

def point_on_segment(x, y, ax, ay, bx, by, eps=1e-10):
    cross = (x - ax) * (by - ay) - (y - ay) * (bx - ax)
    if abs(cross) > eps:
        return False
    return (
        min(ax, bx) - eps <= x <= max(ax, bx) + eps
        and min(ay, by) - eps <= y <= max(ay, by) + eps
    )


def point_in_polygon(x, y, polygon):
    inside = False
    n = len(polygon)
    for i in range(n):
        ax, ay = polygon[i]
        bx, by = polygon[(i + 1) % n]

        if point_on_segment(x, y, ax, ay, bx, by):
            return True

        crosses = ((ay > y) != (by > y))
        if crosses:
            intersection_x = ax + (y - ay) * (bx - ax) / (by - ay)
            if x < intersection_x:
                inside = not inside
    return inside


def classify(lon, lat):
    land = any(point_in_polygon(lon, lat, p) for p in LAND_POLYGONS)
    water_override = any(point_in_polygon(lon, lat, p) for p in WATER_POLYGONS)
    return 1 if land and not water_override else 0


def build_matrix():
    matrix = np.zeros((len(LATS), len(LONS)), dtype=np.uint8)
    for row, lat in enumerate(LATS):
        for col, lon in enumerate(LONS):
            matrix[row, col] = classify(float(lon), float(lat))
    return matrix


def write_outputs(matrix):
    output_dir = Path("outputs")
    output_dir.mkdir(parents=True, exist_ok=True)

    with (output_dir / "candidate.csv").open(
        "w", newline="", encoding="ascii"
    ) as f:
        writer = csv.writer(f, lineterminator="\n")
        for row in matrix:
            writer.writerow([int(value) for value in row])

    # White land, black water.
    image_array = np.where(matrix == 1, 255, 0).astype(np.uint8)
    image = Image.fromarray(image_array, mode="L")
    image.save(output_dir / "candidate.png")

    enlarged = image.resize((len(LONS) * 8, len(LATS) * 8), Image.Resampling.NEAREST)
    enlarged.save(output_dir / "candidate_8x.png")


def validate(matrix):
    assert matrix.shape == (71, 231)
    assert matrix.size == 16401
    assert np.all((matrix == 0) | (matrix == 1))

    with (Path("outputs") / "candidate.csv").open(
        "r", newline="", encoding="ascii"
    ) as f:
        rows = list(csv.reader(f))
    assert len(rows) == 71
    assert all(len(row) == 231 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)

    original = Image.open(Path("outputs") / "candidate.png").convert("L")
    enlarged = Image.open(Path("outputs") / "candidate_8x.png").convert("L")
    assert original.size == (231, 71)
    assert enlarged.size == (1848, 568)
    assert set(original.getdata()).issubset({0, 255})
    assert set(enlarged.getdata()).issubset({0, 255})

    expected = np.repeat(np.repeat(np.asarray(original), 8, axis=0), 8, axis=1)
    assert np.array_equal(expected, np.asarray(enlarged))


def main():
    matrix = build_matrix()
    write_outputs(matrix)
    validate(matrix)


if __name__ == "__main__":
    main()
