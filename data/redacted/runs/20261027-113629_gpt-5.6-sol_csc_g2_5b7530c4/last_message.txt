下面的脚本使用手写近似海岸线多边形和少量椭圆岛屿规则，对每个采样点独立执行点在多边形判定；不是逐点独立推理，也未使用外部地理数据。当前无法在此文本回复中实际运行脚本或嵌入生成后的 PNG，因此不能声称已验证输出。运行后会在 `outputs/` 中生成 CSV 和两张图片。

依赖：Python、NumPy、Pillow。

```python
from pathlib import Path
import csv
import math

import numpy as np
from PIL import Image

# WGS84 sampling grid: north-to-south, west-to-east.
LAT_MAX, LAT_MIN, STEP = 23.0, 1.0, 0.2
LON_MIN, LON_MAX = 99.0, 127.0
lats = LAT_MAX - np.arange(111) * STEP
lons = LON_MIN + np.arange(141) * STEP
land = np.zeros((len(lats), len(lons)), dtype=np.uint8)

# Hand-estimated coastline outlines. Each polygon is a land region.
POLYGONS = [

    # southern China coast and Gulf of Tonkin.

        (108.3, 21.5), (107.5, 21.5), (107.4, 20.5), (106.8, 20.0),

        (103.2, 9.5), (102.2, 9.7), (101.4, 11.0), (100.8, 12.5),
        (100.0, 13.5), (99.5, 15.0), (99.0, 16.0), (99.0, 23.0),
    ],
    # Western Thailand and Myanmar coastline approximation.
    [
        (99.0, 23.0), (101.0, 23.0), (100.5, 21.0), (100.2, 19.0),
        (99.8, 17.0), (99.5, 15.0), (99.2, 13.0), (98.9, 11.0),
        (99.4, 10.0), (100.0, 10.2), (100.2, 12.0), (100.8, 13.5),
        (101.5, 14.5), (102.0, 16.0), (102.2, 18.0), (102.5, 20.0),

    ],
    # Malay peninsula.
    [
        (100.0, 7.0), (100.8, 6.5), (101.5, 5.8), (102.0, 4.8),

    [(108.0, 4.0), (109.0, 4.0), (109.0, 3.0), (108.0, 3.0)],

def point_in_polygon(lon, lat, polygon):
    """Ray-casting test; boundary points are treated as land."""
    inside = False
    n = len(polygon)
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]
        cross = (lon - x1) * (y2 - y1) - (lat - y1) * (x2 - x1)
        if abs(cross) < 1e-10:
            if (min(x1, x2) - 1e-10 <= lon <= max(x1, x2) + 1e-10
                    and min(y1, y2) - 1e-10 <= lat <= max(y1, y2) + 1e-10):
                return True
        if (y1 > lat) != (y2 > lat):
            x_intersect = x1 + (lat - y1) * (x2 - x1) / (y2 - y1)
            if lon < x_intersect:
                inside = not inside
    return inside

# Add small, knowledge-estimated islands as ellipses in longitude/latitude.
# These are deliberately approximate and are not geographic database features.
ELLIPSES = [
    # Spratly islands and reefs, Paracels, and southern South China Sea islets.
    (114.3, 10.5, 0.18, 0.12), (115.0, 10.0, 0.16, 0.11),
    (116.0, 10.8, 0.15, 0.10), (112.9, 8.7, 0.14, 0.10),
    (114.0, 9.0, 0.12, 0.09), (115.8, 9.2, 0.12, 0.09),
    (116.7, 10.4, 0.12, 0.09), (113.8, 16.5, 0.22, 0.13),
    # Gulf of Thailand and Andaman Sea islands.
    (100.0, 9.5, 0.12, 0.25), (99.3, 7.5, 0.10, 0.30),
    # Philippines and Indonesian islands.
    (121.5, 18.0, 0.25, 0.15), (126.0, 7.0, 0.15, 0.20),
    (119.5, 5.5, 0.20, 0.14), (118.0, 1.5, 0.18, 0.14),
]

for row, lat in enumerate(lats):
    for col, lon in enumerate(lons):
        if any(point_in_polygon(float(lon), float(lat), p) for p in POLYGONS):
            land[row, col] = 1
            continue
        for cx, cy, rx, ry in ELLIPSES:
            # Longitude distance is corrected approximately for latitude.
            dx = (lon - cx) * math.cos(math.radians(cy))
            dy = lat - cy
            if (dx / rx) ** 2 + (dy / ry) ** 2 <= 1.0:
                land[row, col] = 1
                break

out_dir = Path("outputs")
out_dir.mkdir(parents=True, exist_ok=True)

with (out_dir / "candidate.csv").open("w", newline="", encoding="ascii") as f:
    writer = csv.writer(f, lineterminator="\n")
    writer.writerows(land.tolist())

# 0 -> black water, 1 -> white land.
image = Image.fromarray((land * 255).astype(np.uint8), mode="L")
image.save(out_dir / "candidate.png")
image.resize((141 * 8, 111 * 8), resample=Image.Resampling.NEAREST).save(
    out_dir / "candidate_8x.png"
)
```
