下面脚本使用手写近似多边形表示加拿大北极群岛主要岛屿、半岛和零散岛屿。每个采样点通过点在多边形内判断，纬度按 79N 到 73N、经度按 105W 到 81W 生成。脚本不联网、不读取外部地图数据。

from pathlib import Path
import csv

import numpy as np
from PIL import Image


# Grid order:
# rows: 79N, 78.9N, ..., 73N
# cols: 105W, 104.9W, ..., 81W
LAT_NORTH = 79.0
LAT_SOUTH = 73.0
LON_WEST = -105.0
LON_EAST = -81.0
STEP = 0.1


def polygon(*points):
    return np.asarray(points, dtype=np.float64)


def inside_polygon(x, y, p):
    """Vectorized point-in-polygon test using the even-odd rule."""
    x0 = p[:, 0, None]
    y0 = p[:, 1, None]
    x1 = np.roll(p[:, 0], -1)[:, None]
    y1 = np.roll(p[:, 1], -1)[:, None]

    yy = y[None, :]
    xx = x[None, :]

    crosses = (y0 > yy) != (y1 > yy)
    denominator = y1 - y0
    denominator = np.where(denominator == 0.0, 1e-30, denominator)

    x_intersection = x0 + (yy - y0) * (x1 - x0) / denominator
    crossings_to_right = crosses & (xx < x_intersection)

    return (np.sum(crossings_to_right, axis=0) % 2) == 1

def build_matrix():
    lats = np.round(
        np.arange(LAT_NORTH, LAT_SOUTH - STEP / 2.0, -STEP),
        10,
    )
    lons = np.round(
        np.arange(LON_WEST, LON_EAST + STEP / 2.0, STEP),
        10,
    )

    if len(lats) != 61 or len(lons) != 241:
        raise RuntimeError("Unexpected grid dimensions")

    lon_grid, lat_grid = np.meshgrid(lons, lats)
    flat_lon = lon_grid.reshape(-1)
    flat_lat = lat_grid.reshape(-1)

    land = np.zeros(flat_lon.shape, dtype=bool)

    for p in LAND_POLYGONS:
        land |= inside_polygon(flat_lon, flat_lat, p)

    matrix = land.astype(np.uint8).reshape(len(lats), len(lons))

    if matrix.shape != (61, 241):
        raise RuntimeError("Matrix is not 61 by 241")

    if matrix.size != 14701:
        raise RuntimeError("Unexpected matrix size")

    if not np.all((matrix == 0) | (matrix == 1)):
        raise RuntimeError("Matrix contains values other than 0 and 1")

    return matrix


def write_outputs(matrix):
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    csv_path = output_dir / "candidate.csv"
    png_path = output_dir / "candidate.png"
    png_8x_path = output_dir / "candidate_8x.png"

    with csv_path.open("w", newline="", encoding="ascii") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # White is land (1), black is water (0).
    image_array = (matrix * 255).astype(np.uint8)
    image = Image.fromarray(image_array, mode="L")

    if image.size != (241, 61):
        raise RuntimeError("Unexpected original image dimensions")

    image.save(png_path, format="PNG")

    resampling = (
        Image.Resampling.NEAREST
        if hasattr(Image, "Resampling")
        else Image.NEAREST
    )
    enlarged = image.resize((241 * 8, 61 * 8), resample=resampling)

    if enlarged.size != (1928, 488):
        raise RuntimeError("Unexpected enlarged image dimensions")

    enlarged.save(png_8x_path, format="PNG")

    # Validate CSV contents after writing.
    with csv_path.open("r", newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))

    if len(rows) != 61:
        raise RuntimeError("CSV does not contain 61 rows")

    if any(len(row) != 241 for row in rows):
        raise RuntimeError("CSV contains a row with an incorrect length")

    if any(value not in {"0", "1"} for row in rows for value in row):
        raise RuntimeError("CSV contains a value other than 0 or 1")

    # Validate that both PNGs contain only black and white.
    original_values = set(np.unique(np.asarray(image)).tolist())
    enlarged_values = set(np.unique(np.asarray(enlarged)).tolist())

    if not original_values.issubset({0, 255}):
        raise RuntimeError("Original PNG is not strictly black and white")

    if not enlarged_values.issubset({0, 255}):
        raise RuntimeError("Enlarged PNG is not strictly black and white")

    return csv_path, png_path, png_8x_path


def main():
    matrix = build_matrix()
    paths = write_outputs(matrix)

    print("matrix:", matrix.shape, "values:", matrix.size)
    for path in paths:
        print(path)


if __name__ == "__main__":
    main()
