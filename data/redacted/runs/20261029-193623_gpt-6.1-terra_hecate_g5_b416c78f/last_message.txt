def contains(polygon, x, y):
    """Even-odd point-in-polygon test, including boundary points."""
    inside = False
    previous = polygon[-1]
    for current in polygon:
        x1, y1 = previous
        x2, y2 = current
        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if (
            abs(cross) < 1e-12
            and min(x1, x2) <= x <= max(x1, x2)
            and min(y1, y2) <= y <= max(y1, y2)
        ):
            return True
        if (y1 > y) != (y2 > y):
            intersection_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < intersection_x:
                inside = not inside
        previous = current
    return inside


def main():
    OUTPUT.mkdir(parents=True, exist_ok=True)

    matrix = []
    for row in range(SIZE):
        latitude = 55.0 - 2.0 * row / (SIZE - 1)
        values = []
        for col in range(SIZE):
            longitude = -131.0 + 2.0 * col / (SIZE - 1)
            values.append(
                int(any(contains(poly, longitude, latitude) for poly in LAND))
            )
        matrix.append(values)

    csv_path = OUTPUT / "candidate.csv"
    with csv_path.open("w", newline="", encoding="utf-8") as f:
        csv.writer(f).writerows(matrix)

    image = Image.new("L", (SIZE, SIZE))
    image.putdata([255 * value for row in matrix for value in row])
    image.save(OUTPUT / "candidate.png")

    resampling = getattr(Image, "Resampling", Image)
    enlarged = image.resize((SIZE * SCALE, SIZE * SCALE), resampling.NEAREST)
    enlarged.save(OUTPUT / "candidate_8x.png")

        saved = list(csv.reader(f))
    assert len(saved) == SIZE
    assert all(len(row) == SIZE for row in saved)
    assert all(value in ("0", "1") for row in saved for value in row)
    assert [[int(value) for value in row] for row in saved] == matrix

    with Image.open(OUTPUT / "candidate.png") as original:
        original.load()
        assert original.size == (SIZE, SIZE)
        assert set(original.getdata()) <= {0, 255}
        assert list(original.getdata()) == [

            large.load()
            assert large.size == (SIZE * SCALE, SIZE * SCALE)
            assert set(large.getdata()) <= {0, 255}
            pixels = original.load()
            large_pixels = large.load()
            assert all(
                large_pixels[x, y] == pixels[x // SCALE, y // SCALE]
                for y in range(SIZE * SCALE)
                for x in range(SIZE * SCALE)
            )

    print("Validated 127 x 127 matrix: 16,129 binary values.")
    print("Validated binary PNGs and exact 8x nearest-neighbor enlargement.")
    print("Files written to:", OUTPUT.resolve())


if __name__ == "__main__":
    main()
