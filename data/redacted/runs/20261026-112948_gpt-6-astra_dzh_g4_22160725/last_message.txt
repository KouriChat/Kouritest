def inside_polygon(lon_grid, lat_grid, polygon):
    """Vectorized point-in-polygon test; boundary points count as land."""
    inside = np.zeros(lon_grid.shape, dtype=bool)
    n = len(polygon)
    eps = 1e-10

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        dx = x2 - x1
        dy = y2 - y1
        cross = (lon_grid - x1) * dy - (lat_grid - y1) * dx
        on_segment = (
            (np.abs(cross) <= eps)
            & (lon_grid >= min(x1, x2) - eps)
            & (lon_grid <= max(x1, x2) + eps)
            & (lat_grid >= min(y1, y2) - eps)
            & (lat_grid <= max(y1, y2) + eps)
        )
        inside |= on_segment

        crosses_ray = (y1 > lat_grid) != (y2 > lat_grid)
        x_intersection = x1 + (lat_grid - y1) * dx / (dy if dy != 0 else 1.0)
        inside ^= crosses_ray & (lon_grid < x_intersection)

    return inside

    land = np.zeros((len(latitudes), len(longitudes)), dtype=bool)
    for polygon in MAINLAND + ISLANDS:
        land |= inside_polygon(lon_grid, lat_grid, polygon)

    matrix = land.astype(np.uint8)

        writer = csv.writer(f, lineterminator="\n")
        writer.writerows(matrix.tolist())

    # Mode 1 maps land to white and water to black.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(output_dir / "candidate.png")
    enlarged = image.resize((1848, 568), resample=Image.Resampling.NEAREST)
