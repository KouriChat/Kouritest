def on_land(lon, lat):
    for polygon in POLYGONS:
        inside = False
        j = len(polygon) - 1
        for i, (xi, yi) in enumerate(polygon):
            xj, yj = polygon[j]
            # Ray crossing; boundary points count as land.
            cross = (yi > lat) != (yj > lat)
            if cross:
                edge_x = (xj - xi) * (lat - yi) / (yj - yi) + xi
                if lon < edge_x:
                    inside = not inside
            j = i
        if inside:
            return 1
    return 0


def main():
    out = Path("outputs")
    out.mkdir(parents=True, exist_ok=True)

    lats = [23.0 - 0.2 * r for r in range(111)]
    lons = [99.0 + 0.2 * c for c in range(141)]
    matrix = np.array(
        [[on_land(lon, lat) for lon in lons] for lat in lats],
        dtype=np.uint8,
    )

    if matrix.shape != (111, 141) or matrix.size != 15651:
        raise RuntimeError("Unexpected matrix dimensions")
    if not np.isin(matrix, (0, 1)).all():
        raise RuntimeError("Matrix contains values other than 0 and 1")

    with (out / "candidate.csv").open("w", encoding="ascii", newline="") as f:
        for row in matrix:
            f.write(",".join(str(int(v)) for v in row) + "\n")

    # 0 is black water; 1 is white land.
    image = Image.fromarray(matrix * 255, mode="L")
    image.save(out / "candidate.png")
    enlarged = image.resize((1128, 888), resample=Image.Resampling.NEAREST)
    enlarged.save(out / "candidate_8x.png")

    # Verify the saved artifacts and nearest-neighbor replication.
    rows = (out / "candidate.csv").read_text(encoding="ascii").splitlines()
    if len(rows) != 111 or any(
        len(row.split(",")) != 141 or set(row.split(",")) - {"0", "1"}
        for row in rows
    ):
        raise RuntimeError("CSV validation failed")
    if image.size != (141, 111) or set(image.getdata()) - {0, 255}:
        raise RuntimeError("Source PNG validation failed")
    if enlarged.size != (1128, 888):
        raise RuntimeError("Enlarged PNG dimensions are incorrect")
    enlarged_arr = np.asarray(enlarged)
    if not np.array_equal(enlarged_arr, np.repeat(np.repeat(matrix * 255, 8, axis=0), 8, axis=1)):
        raise RuntimeError("Enlarged PNG is not an exact nearest-neighbor scale")

    print(f"Validated {matrix.shape[0]}x{matrix.shape[1]} matrix, {matrix.size} values.")
    print("Wrote outputs/candidate.csv, outputs/candidate.png, outputs/candidate_8x.png")


if __name__ == "__main__":
    main()
