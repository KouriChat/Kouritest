def polygon_contains(lon, lat, polygon):
    """Ray-casting point-in-polygon test; boundary points count as land."""
    inside = False
    j = len(polygon) - 1

    for i in range(len(polygon)):
        xi, yi = polygon[i]
        xj, yj = polygon[j]

        cross = (lon - xi) * (yj - yi) - (lat - yi) * (xj - xi)
        if abs(cross) < 1e-12:
            if (
                min(xi, xj) - 1e-12 <= lon <= max(xi, xj) + 1e-12
                and min(yi, yj) - 1e-12 <= lat <= max(yi, yj) + 1e-12
            ):
                return True

        if (yi > lat) != (yj > lat):
            edge_lon = (xj - xi) * (lat - yi) / (yj - yi) + xi
            if lon < edge_lon:
                inside = not inside

        j = i

    return inside


def ellipse(cx, cy, rx, ry, count=24):
    return [
        (
            cx + rx * math.cos(2.0 * math.pi * i / count),
            cy + ry * math.sin(2.0 * math.pi * i / count),
        )
        for i in range(count)
    ]

def is_land(lon, lat):
    if not any(polygon_contains(lon, lat, p) for p in land_polygons):
        return False
    return not any(polygon_contains(lon, lat, p) for p in water_holes)


def build_matrix():
    matrix = np.zeros((ROWS, COLS), dtype=np.uint8)

    for row in range(ROWS):
        lat = MAX_LAT - row * STEP
        for col in range(COLS):
            lon = MIN_LON + col * STEP
            matrix[row, col] = 1 if is_land(lon, lat) else 0

    return matrix


def write_outputs(matrix):
    output_dir = Path.cwd() / "outputs"
    output_dir.mkdir(parents=True, exist_ok=True)

    assert matrix.shape == (64, 253)
    assert matrix.size == 16192
    assert np.all((matrix == 0) | (matrix == 1))

    csv_path = output_dir / "candidate.csv"
    with csv_path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle, lineterminator="\n")
        for row in matrix:
            writer.writerow([int(value) for value in row])

    pixels = np.where(matrix == 1, 255, 0).astype(np.uint8)

    original = Image.fromarray(pixels, mode="L")
    original_path = output_dir / "candidate.png"
    original.save(original_path)

    enlarged = original.resize(
        (COLS * 8, ROWS * 8),
        resample=Image.Resampling.NEAREST
        if hasattr(Image, "Resampling")
        else Image.NEAREST,
    )
    enlarged_path = output_dir / "candidate_8x.png"
    enlarged.save(enlarged_path)

    assert original.size == (253, 64)
    assert enlarged.size == (2024, 512)
    assert set(np.unique(np.asarray(original))).issubset({0, 255})
    assert set(np.unique(np.asarray(enlarged))).issubset({0, 255})

    expected_enlarged = np.repeat(np.repeat(pixels, 8, axis=0), 8, axis=1)
    assert np.array_equal(np.asarray(enlarged), expected_enlarged)

    with csv_path.open("r", encoding="utf-8") as handle:
        csv_rows = list(csv.reader(handle))

    assert len(csv_rows) == 64
    assert all(len(row) == 253 for row in csv_rows)
    assert all(value in {"0", "1"} for row in csv_rows for value in row)

    print(f"matrix: {matrix.shape}, valid values: {matrix.size}")
    print(f"csv: {csv_path}")
    print(f"png: {original_path} ({original.size[0]}x{original.size[1]})")
    print(f"png: {enlarged_path} ({enlarged.size[0]}x{enlarged.size[1]})")


if __name__ == "__main__":
    write_outputs(build_matrix())
