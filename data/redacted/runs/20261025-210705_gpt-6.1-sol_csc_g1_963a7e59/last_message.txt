def point_in_polygon(x, y, polygon):
    """射线法判断点是否在多边形内；边界按陆地处理。"""
    inside = False
    n = len(polygon)

    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]

        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if abs(cross) < 1e-10:
            if (
                min(x1, x2) - 1e-10 <= x <= max(x1, x2) + 1e-10
                and min(y1, y2) - 1e-10 <= y <= max(y1, y2) + 1e-10
            ):
                return True

        if (y1 > y) != (y2 > y):
            xinters = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < xinters:
                inside = not inside

    return inside


def in_ellipse(lon, lat, cx, cy, rx, ry):
    return ((lon - cx) / rx) ** 2 + ((lat - cy) / ry) ** 2 <= 1.0

def is_land(lon, lat):
    for polygon in POLYGONS:
        if point_in_polygon(lon, lat, polygon):
            return 1

    for cx, cy, rx, ry in ELLIPSES:
        if in_ellipse(lon, lat, cx, cy, rx, ry):
            return 1

    return 0


def generate_matrix():
    matrix = np.zeros((N_ROWS, N_COLS), dtype=np.uint8)

    for row in range(N_ROWS):
        lat = LAT_MAX - STEP * row
        for col in range(N_COLS):
            lon = LON_MIN + STEP * col
            matrix[row, col] = is_land(lon, lat)

    return matrix


def write_csv(matrix):
    path = OUT_DIR / "candidate.csv"
    with path.open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f, lineterminator="\n")
        for row in matrix:
            writer.writerow([int(v) for v in row])


def write_images(matrix):
    # 水域为黑色 0，陆地为白色 255。
    raw = (matrix * 255).astype(np.uint8)
    raw_image = Image.fromarray(raw, mode="L")
    raw_image.save(OUT_DIR / "candidate.png")

    # 直接重复像素，等价于最近邻放大 8 倍，不做平滑或修补。
    enlarged = np.repeat(np.repeat(raw, 8, axis=0), 8, axis=1)
    enlarged_image = Image.fromarray(enlarged, mode="L")
    enlarged_image.save(OUT_DIR / "candidate_8x.png")


def validate(matrix):
    assert matrix.shape == (111, 141)
    assert matrix.size == 15651
    assert np.all((matrix == 0) | (matrix == 1))

    with (OUT_DIR / "candidate.csv").open("r", encoding="utf-8") as f:
        rows = list(csv.reader(f))

    assert len(rows) == 111
    assert all(len(row) == 141 for row in rows)
    assert all(value in {"0", "1"} for row in rows for value in row)

    raw = np.asarray(Image.open(OUT_DIR / "candidate.png").convert("L"))
    enlarged = np.asarray(
        Image.open(OUT_DIR / "candidate_8x.png").convert("L")
    )

    assert raw.shape == (111, 141)
    assert enlarged.shape == (888, 1128)
    assert set(np.unique(raw)).issubset({0, 255})
    assert set(np.unique(enlarged)).issubset({0, 255})

    expected = np.repeat(np.repeat(raw, 8, axis=0), 8, axis=1)
    assert np.array_equal(expected, enlarged)


def main():
    matrix = generate_matrix()
    write_csv(matrix)
    write_images(matrix)
    validate(matrix)

    land_count = int(matrix.sum())
    water_count = int(matrix.size - land_count)

    print("生成完成")
    print(f"矩阵尺寸: {matrix.shape[0]} x {matrix.shape[1]}")
    print(f"有效值数量: {matrix.size}")
    print(f"陆地点: {land_count}")
    print(f"水域点: {water_count}")
    print("输出文件:")
    print("  outputs/candidate.csv")
    print("  outputs/candidate.png")
    print("  outputs/candidate_8x.png")


if __name__ == "__main__":
    main()
