def contains(poly, x, y):
    """Even-odd point-in-polygon test, including points on edges."""
    inside = False
    previous = poly[-1]
    for current in poly:
        x1, y1 = previous
        x2, y2 = current
        cross = (x - x1) * (y2 - y1) - (y - y1) * (x2 - x1)
        if (
            abs(cross) < 1e-10
            and min(x1, x2) - 1e-10 <= x <= max(x1, x2) + 1e-10
            and min(y1, y2) - 1e-10 <= y <= max(y1, y2) + 1e-10
        ):
            return True
        if (y1 > y) != (y2 > y):
            crossing_x = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            if x < crossing_x:
                inside = not inside
        previous = current
    return inside


def main():
    output = Path("outputs")
    output.mkdir(parents=True, exist_ok=True)

    # Integer arithmetic defines the sampling coordinates exactly to 0.2 deg.
    matrix = [
        [
            int(any(contains(poly, (-45 + col) / 5, (225 - row) / 5)
                    for poly in LAND_POLYGONS))
            for col in range(231)
        ]
        for row in range(71)
    ]

    csv_path = output / "candidate.csv"
    with csv_path.open("w", newline="", encoding="ascii") as f:
        csv.writer(f, lineterminator="\n").writerows(matrix)

    image = Image.new("L", (231, 71))
    image.putdata([255 * value for row in matrix for value in row])
    image.save(output / "candidate.png")

    resampling = getattr(Image, "Resampling", Image)
    enlarged = image.resize((1848, 568), resample=resampling.NEAREST)
    enlarged.save(output / "candidate_8x.png")

    # Validate the actual files, not only the in-memory matrix.
    with csv_path.open(newline="", encoding="ascii") as f:
        rows = list(csv.reader(f))
    assert len(rows) == 71
    assert all(len(row) == 231 for row in rows)
    assert all(value in ("0", "1") for row in rows for value in row)
    assert sum(len(row) for row in rows) == 16401

    with Image.open(output / "candidate.png") as original:
        original.load()
        assert original.size == (231, 71)
        assert set(original.getdata()) <= {0, 255}
        assert list(original.getdata()) == [
            255 * int(value) for row in rows for value in row
        ]
        expected = original.resize((1848, 568), resample=resampling.NEAREST)
        with Image.open(output / "candidate_8x.png") as scaled:
            scaled.load()
            assert scaled.size == (1848, 568)
            assert set(scaled.getdata()) <= {0, 255}
            assert scaled.tobytes() == expected.tobytes()

    print("Validated: 71 x 231 = 16401 values; both PNG files are binary.")
    print("Output: outputs/candidate.csv")
    print("Output: outputs/candidate.png")
    print("Output: outputs/candidate_8x.png")


if __name__ == "__main__":
    main()
